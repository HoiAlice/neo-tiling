\section{The least $\ell_r$ perturbation: an exact algorithm}\label{sec:lp}

\subsection{Problem}
An index $t\in[T]$ is called a \emph{year}. New prices are written $q_t=p_t\circ(1+\varepsilon_t)$ with $\varepsilon_t>-1$ coordinatewise, and the deviation of $q$ from $p$ is the relative price error $\dev(q,p):=q/p-1$, coordinatewise. This is the price-error variant of the Afriat--Varian indices \citep{Varian1985}; see \citet{DemuynckRehbeck2023}. For weights $c^i_t\ge0$ and $1\le r<\infty$,
\[
C_r(Q):=\norm{c\circ\dev(Q,P)}_r,\qquad\OPT_r:=\inf\set{C_r(Q)\mid Q>0,\ y\text{ weakly solvable at }Q}.
\]

To half-widths $\alpha,\beta\in\Rp[d\times T]$ with $\alpha\le1$ we associate the box with the lower and upper corners $L_t:=p_t\circ(1-\alpha_t)\ge0$, $U_t:=p_t\circ(1+\beta_t)$ and the corner family $W(\alpha,\beta):=(L,U)$. The query $\BOX(\alpha,\beta)$ is the box query of \cref{sec:corners} on $W(\alpha,\beta)$ with $\mathrm{lo}=(y,0)$, $\mathrm{hi}=(\infty,y)$: some $w\in\cone\set{\ind S\mid S\in\wSp(W(\alpha,\beta))}$ has $w^-_t\ge y_t\ge w^+_t$ for all $t$.

\begin{definition}[box feasibility]\label{def:boxfeasible}
The half-widths $(\alpha,\beta)$ are \emph{feasible} if $\BOX(\alpha,\beta)$ holds. The \emph{box cost} of $(\alpha,\beta)$ is $\norm{c\circ\max(\alpha,\beta)}_r$, the maximum taken coordinatewise.
\end{definition}
A point $q$ of the box has $-\alpha\le\dev(q,p)\le\beta$, so the box cost bounds $C_r(Q)$ for every $Q$ in the box.

\paragraph{Literature.} The goodness-of-fit indices of revealed preference perturb one scalar per observation \citep{Cherchye2014}. For GARP the $\ell_1$ perturbation of every price is a single MILP \citep{DemuynckRehbeck2023}. Here the witnesses are latent and weak solvability is NP-hard for $d\ge3$ (\cref{sec:hardness}), so the problem splits into a master problem and an exact oracle, as in \citet{DeanMartin2016}. The master is a monotonic optimization problem \citep{Tuy2000,TuyAlKhayyalThach2005}, and the cuts from conflicts are combinatorial Benders cuts \citep{CodatoFischetti2006,FischettiSalvagninZanette2010}.

\subsection{Reduction to boxes}
\begin{theorem}[reduction]\label{thm:reduction}
Let $d\ge2$, $1\le r<\infty$ and $c\ge0$. For every $C$ there is a positive $Q$ with $y$ weakly solvable at $Q$ and $C_r(Q)\le C$ if and only if there are feasible half-widths $\alpha,\beta\ge0$, $\alpha<1$, with box cost at most $C$. Hence
\[
\OPT_r=\inf\set{\norm{c\circ\max(\alpha,\beta)}_r\mid\alpha,\beta\ge0,\ \alpha<1,\ \BOX(\alpha,\beta)}.
\]
The feasible half-widths are closed upward: if $\BOX(\alpha,\beta)$, $\alpha\le\alpha'\le1$ and $\beta\le\beta'$, then $\BOX(\alpha',\beta')$.
\end{theorem}
\begin{proof}
($\Rightarrow$) Let $\varepsilon:=\dev(Q,P)$, $\alpha:=\varepsilon^-$ and $\beta:=\varepsilon^+$. Since $q>0$, $\alpha<1$. The box contains $Q$, so the corner theorem (\cref{thm:corners}) gives $\BOX(\alpha,\beta)$, and $\max(\varepsilon^-,\varepsilon^+)=\abs\varepsilon$, so the box cost equals $C_r(Q)$. ($\Leftarrow$) By \cref{thm:onesigned}(a) the box contains a positive $Q$ at which $y$ is weakly solvable and $\abs{\dev(Q,P)}\le\max(\alpha,\beta)$. The infimum formula follows.

Upward closedness: by \cref{thm:corners} a feasible box contains a $Q$ at which $y$ is weakly solvable, and $Q$ lies in every larger box; the condition $\alpha'\le1$ keeps the lower corners nonnegative.

\emph{Observation (deleting years).} For $Y\subset[T]$ let $\BOX_Y$ be the box query for the years in $Y$ only, with the prices, outputs and corners of these years. If $\BOX(\alpha,\beta)$ holds, then so does $\BOX_Y$ for the restrictions of $\alpha,\beta$ to $Y$: a weakly reachable set of all corners, intersected with the corners of $Y$, is weakly reachable there with the same witnesses, and restricting $w$ to these corners keeps the bounds. Hence a NO of $\BOX_Y$ is a NO of $\BOX(\alpha',\beta')$ for all half-widths $(\alpha',\beta')$ that agree with $(\alpha,\beta)$ on $Y$.
\end{proof}

\subsection{One-signedness}
A positive $Q$ is \emph{one-signed} if every year moves in one direction: for each $t$ either $q_t\le p_t$ or $q_t\ge p_t$.
\begin{theorem}[one-signedness]\label{thm:onesigned}
Let $d\ge2$.
\begin{enumerate}[label=(\alph*)]
\item If $\alpha,\beta\ge0$, $\alpha<1$, and $\BOX(\alpha,\beta)$, then there is a one-signed positive $Q$ with $y$ weakly solvable at $Q$ and $-\alpha\le\dev(Q,P)\le\beta$, so $\abs{\dev(Q,P)}\le\max(\alpha,\beta)$.
\item For every positive $Q$ with $y$ weakly solvable at $Q$ there is a one-signed positive $Q'$ with $y$ weakly solvable at $Q'$ and $\abs{\dev(Q',P)}\le\abs{\dev(Q,P)}$.
\end{enumerate}
In particular $\OPT_r$ is the infimum over one-signed $Q$.
\end{theorem}
\begin{proof}
(a) The path $s\mapsto p_t\circ\bigl(1-\alpha_t\max(1-2s,0)+\beta_t\max(2s-1,0)\bigr)$, $s\in[0,1]$, runs from $L_t$ through $p_t$ to $U_t$. It is continuous, and positive because $\alpha<1$. The corner theorem (\cref{thm:corners}) gives a point $q_t$ of the path for each $t$ such that $y$ is weakly solvable at $Q$. If $s_t\le1/2$, then $q_t\le p_t$; otherwise $q_t\ge p_t$. (b) The box with half-widths $(\varepsilon^-,\varepsilon^+)$, $\varepsilon:=\dev(Q,P)$, contains $Q$ and is feasible by \cref{thm:reduction}; apply (a).
\end{proof}
The half-widths $(\varepsilon^-,\varepsilon^+)$ of a one-signed $Q$ have $\alpha_t=0$ or $\beta_t=0$ in every year. On such half-widths $\max(\alpha,\beta)=\alpha+\beta$, so the box cost is $\norm{c\circ(\alpha+\beta)}_r$, the objective of the master problem below.

\subsection{Covers and the certificate of infeasibility}\label{sec:covers}
Covers present in a family and the sets respecting them are defined in \cref{lem:cover}.
\begin{proposition}[cover]\label{prop:cover}
Let $d\ge2$, let $y\ge0$ be weakly solvable at $Q\ge0$, and let $(t,A)$ be a cover present in $Q$. Then $y_t\le\sum_{s\in A}y_s$. In particular $q_s<q_t$ implies $y_t\le y_s$.
\end{proposition}
\begin{proof}
\Cref{thm:closure} puts $y$ in the cone of the indicators of the sets in $\wSp(Q)$. Each of them respects the cover $(t,A)$ by \cref{lem:cover}, so the functional $x\mapsto\sum_{s\in A}x_s-x_t$ is $\ge0$ on every generator. Dominance is the case $A=\set{s}$.
\end{proof}

The exact box oracle of \cref{sec:corners} answers NO with a certificate $(\lambda,\mathcal C)$, $\lambda\in\Q^{2T}$, satisfying (F1) and (F2) of \cref{prop:oracle} for the bounds $\mathrm{lo}=(y,0)$, $\mathrm{hi}=(\infty,y)$ of the box query. Condition (F2) is a 0/1 problem with integer weights, and the bounds of the query depend on $y$ only.
\begin{theorem}[certificate]\label{thm:certificate}
Let $(\lambda,\mathcal C)$ satisfy (F1), (F2) of \cref{prop:oracle} for the bounds of the box query, and let $W'$ be a family of $2T$ points in which all covers of $\mathcal C$ are present. Then no $w\in\cone\set{\ind S\mid S\in\wSp(W')}$ satisfies the bounds of the box query. In particular, if $W'=W(\alpha',\beta')$, then $\BOX(\alpha',\beta')$ fails.
\end{theorem}
\begin{proof}
By \cref{lem:cover} every $S\in\wSp(W')$ respects $\mathcal C$, so $\lambda(S)\ge0$ by (F2). Hence $\ip\lambda w\ge0$ for every $w$ of the cone. By (F1) with $0\cdot\infty=0$, $\lambda\le0$ on the lower corners. If $w^-_t\ge y_t\ge w^+_t$ for all $t$, then, as $w\ge0$, $\ip\lambda w\le-\sum_t\lambda^-_{L_t}y_t+\sum_t\lambda^+_{U_t}y_t<0$ by (F1), a contradiction.
\end{proof}

\paragraph{Minimal conflicts and cover cuts.}\label{sec:conflicts}
A deletion filter preserving (F2) \citep{CodatoFischetti2006} reduces $\mathcal C$ to an inclusion-minimal $\mathcal C'$. By \cref{thm:certificate} every feasible box makes some cover of $\mathcal C'$ absent; for the combination $\mu$ of the certificate, some coordinate $i$ has $\sum_{s\in A}\mu_sW^i_s\ge W^i_t$. The corners are linear in $z=(\alpha,\beta)$, so the cut
\[
\bigvee_{(t,A)\in\mathcal C'}\ \bigvee_{i\in[d]}\Bigl[\sum_{s\in A}\mu_sW^i_s(z)\ge W^i_t(z)\Bigr]
\]
is a valid disjunction of linear inequalities. Covers with $\abs A=1$ are dominances (\cref{prop:cover}).

\subsection{Algorithm}\label{sec:algo}
\paragraph{The abstract method.}
Let $\iota$ be a finite index set, $M_j>0$ for $j\in\iota$, $B:=\prod_j[0,M_j]$ and $\mathbf M$ its top corner. Let $H$ be up-closed in $B$ with $\mathbf M\in H$, and let $O$ be a membership oracle for $H$ on $B$. The cost $f\ge0$ is monotone with $f(z+a\e)\le f(z)+aL$ for $a\ge0$ and a constant $L>0$. A class of \emph{one-signed} points represents $H$: every $z\in H$ has a one-signed $z'\in H$ with $f(z')\le f(z)$. For a list $C$ of cut points let $\mathrm{Sat}(C):=\set{z\mid\forall\hat z\in C\ \exists j\colon z_j>\hat z_j}$ (polyblock cuts, strict). Given $C$ such that $\mathrm{Sat}(C)\cap B$ contains a one-signed point, and a number $\UB$, the master oracle returns $v\in\mathrm{Sat}(C)\cap B$ and $\mathrm{lb}$ with $\mathrm{lb}\le f(z)$ for every one-signed $z\in\mathrm{Sat}(C)\cap B$ and, for $0<\UB$ and $0\le\LB\le\UB$, with the relative gap $f(v)-\mathrm{lb}\le\frac{\UB-\LB}{4\UB}\,f(v)$ of a MILP solver (the master receives $C$, $\UB$ and $\LB$). Put $\OPT:=\inf f(H)$. A state $(C,\UB,\LB,\mathrm{best})$ is \emph{stopped} for the stopping tolerance $\eta>0$ if $\UB-\LB\le\eta\UB$. Step~0 queries $O(0)$. On YES the state is $(\varnothing,f(0),f(0),0)$, and on NO it is $([0],f(\mathbf M),0,\mathbf M)$. A step leaves a stopped state unchanged. Otherwise it calls the master on $(C,\UB,\LB)$, puts $\tau:=(\UB-\LB)/(2L)$ and $w:=\min(v+\tau\e,\mathbf M)$ coordinatewise, and queries $O(w)$. On YES it sets $\UB\leftarrow\min(\UB,f(w))$ and updates $\mathrm{best}$. On NO it appends $w$ to $C$. In both cases $\LB\leftarrow\max(\LB,\mathrm{lb})$.

For the box instance, $\iota$ is two copies of $[T]\times[d]$, $z=(\alpha,\beta)$, $\mathbf M=(\mathbf M^\alpha,\mathbf M^\beta)$, $H=\set{z\in B\mid\BOX(\alpha,\beta)}$, $O$ is the exact box oracle and $f$ is the box cost. For $\mathbf M^\alpha\le1$ the set $H$ is up-closed by \cref{thm:reduction}. Since $\max(\alpha+a\e,\beta+a\e)=\max(\alpha,\beta)+a\e$, Minkowski's inequality gives $L=\norm c_r$. A point is one-signed if $\alpha_t=0$ or $\beta_t=0$ for each $t$. For $\mathbf M^\alpha<1$ these points represent $H$: for $z\in H$, \cref{thm:onesigned}(a) gives a one-signed $Q$ in the box, and $z':=(\varepsilon^-,\varepsilon^+)\le z$ with $\varepsilon:=\dev(Q,P)$ is one-signed, lies in $H$ by \cref{thm:reduction}, and has $f(z')\le f(z)$.

\begin{lemma}[nonemptiness]\label{lem:nonempty}
Let $i_0\in[d]$ and $\underline p:=\min_sp^{i_0}_s$. Define $Q$ by $q^{i_0}_t:=\underline p$ and $q^i_t:=p^i_t$ for $i\ne i_0$. Then every $y\in\Rp[T]$ is weakly solvable at $Q$. Hence $\BOX(\alpha,\beta)$ holds for $\beta=0$ and $\alpha^{i_0}_t:=1-\underline p/p^{i_0}_t<1$, $\alpha^i_t:=0$ for $i\ne i_0$, and for all larger half-widths with $\alpha\le1$.
\end{lemma}

\begin{proof}
The unit vector $e_{i_0}$ is a weak witness of every $t$ for every $S\subset[T]$, since $\ip{e_{i_0}}{q_s-q_t}=0$. So $y=\sum_ty_t\ind{\set t}$ lies in the cone of \cref{thm:closure}, and $Q>0$. The box with these half-widths contains $Q$, so $\BOX$ holds by \cref{thm:corners}, and for larger half-widths by \cref{thm:reduction}.
\end{proof}

Thus all conditions on $H$ and $f$ hold for the box instance whenever $c>0$, $\mathbf M^\alpha<1$ and the top corner $\mathbf M$ is feasible, which holds by \cref{lem:nonempty} if $\mathbf M^\alpha$ dominates the half-widths of that lemma for some $i_0$. The only assumptions that remain external are the two oracle specifications, that is, the correctness of the box oracle and of the master solver.

\begin{theorem}[correctness and termination]\label{thm:algo}
Let the membership oracle $O$ and the master oracle meet their specifications. For the box instance let moreover $d\ge2$, $T\ge1$, $1\le r<\infty$, $c>0$, $\mathbf M>0$, $\mathbf M^\alpha<1$, and let $\mathbf M^\alpha$ dominate the half-widths of \cref{lem:nonempty} for some $i_0\in[d]$, so that the aligned prices lie in the box.
\begin{enumerate}[label=(\alph*)]
\item At every iteration $\LB\le\OPT\le\UB=f(\mathrm{best})$ with $\mathrm{best}\in H$. For the box instance there is a positive $Q$ with $y$ weakly solvable at $Q$ and $C_r(Q)\le\UB$, and $\LB\le C_r(Q)$ for every positive $Q$ at which $y$ is weakly solvable and $\varepsilon^-\le\mathbf M^\alpha$, $\varepsilon^+\le\mathbf M^\beta$ for $\varepsilon:=\dev(Q,P)$.
\item Let $\OPT>0$ and $\eta>0$. The method reaches a stopped state after at most
\[
\Bigl\lceil\log_{4/3}\frac{f(\mathbf M)}{\eta\OPT}\Bigr\rceil+1+\Bigl(\Bigl\lfloor\frac{2LM_{\max}}{\eta\OPT}\Bigr\rfloor+1\Bigr)^{\abs\iota}
\]
iterations, where $M_{\max}:=\max_jM_j$; for the box instance $\abs\iota=2dT$ and $L=\norm c_r$.
\end{enumerate}
\end{theorem}
Since $c^i_t\abs{\varepsilon^i_t}\le C_r(Q)$, the choice $\mathbf M^\beta\ge\UB_0/c$ and $\UB_0/c\le\mathbf M^\alpha<1$ on the $\alpha$-sides, for a known feasible cost $\UB_0$, gives $\LB\le\OPT_r$ whenever $\UB_0<\min_{i,t}c^i_t$ (not formalized).

\begin{proof}
(a) Call a state \emph{good} if $\LB\le\OPT$, $\mathrm{best}\in H$, $f(\mathrm{best})=\UB$, and every cut point lies in $B\setminus H$. In a good state every $z\in H$ lies in $\mathrm{Sat}(C)$: otherwise $z\le\hat z\in B$ for a cut point $\hat z$, and $\hat z\in H$ by up-closedness. In particular the one-signed representative of $\mathbf M$ lies in $\mathrm{Sat}(C)\cap B$, so the master is called in its domain. For $z\in H$ its one-signed representative $z'\in H\subset\mathrm{Sat}(C)$ gives $\mathrm{lb}\le f(z')\le f(z)$, hence $\mathrm{lb}\le\OPT$. Step~0 gives a good state: on YES, $f(z)\ge f(0)$ for all $z\in B$ by monotonicity; on NO, $\mathbf M\in H$ and $0\notin H$. A step preserves goodness: $\max(\LB,\mathrm{lb})\le\OPT$, a YES point $w$ lies in $H$, and a NO point $w$ lies in $B\setminus H$. By induction every state is good, and $\OPT\le f(\mathrm{best})=\UB$. For the box instance, $\mathrm{best}$ has $\alpha\le\mathbf M^\alpha<1$ and passes $\BOX$, so ($\Leftarrow$) of \cref{thm:reduction} gives $Q$ with $C_r(Q)\le\UB$. Conversely, for $Q$ as in the statement $z:=(\varepsilon^-,\varepsilon^+)$ lies in $H$ by ($\Rightarrow$) of \cref{thm:reduction}, and $\LB\le\OPT\le f(z)=C_r(Q)$.

(b) The gap $\UB-\LB$ never grows, since $\UB$ does not increase and $\LB$ does not decrease. If $O(0)=\mathrm{YES}$, step~0 is stopped; so let $O(0)=\mathrm{NO}$, with initial gap $f(\mathbf M)$. Suppose the method is not stopped up to the bound.

\emph{YES steps contract the gap by $3/4$.} By monotonicity and the shift property, $f(w)\le f(v+\tau\e)\le f(v)+\frac12(\UB-\LB)$. Put $g:=\frac{\UB-\LB}{4\UB}$ and $\LB':=\max(\LB,\mathrm{lb})$. If $f(v)\le\UB$, the relative gap gives $f(v)-\LB'\le g\,f(v)\le\frac14(\UB-\LB)$, so the new gap is at most $f(w)-\LB'\le\frac34(\UB-\LB)$. If $f(v)>\UB$, then $\mathrm{lb}\ge(1-g)f(v)\ge(1-g)\UB=\UB-\frac14(\UB-\LB)$ (here $g\le\frac14$ because $\LB\ge0$), so the new lower bound alone leaves a gap of at most $\frac14(\UB-\LB)$, whatever the query answers. A non-stopped state has $\UB-\LB>\eta\UB\ge\eta\OPT$. After $Y$ YES steps $\eta\OPT<(3/4)^Yf(\mathbf M)$, so $Y<\log_{4/3}(f(\mathbf M)/(\eta\OPT))$.

\emph{NO points are separated.} Put $\tau_0:=\eta\OPT/(2L)$; a non-stopped state has $\tau>\tau_0$. Let step $m$ answer NO and let $n>m$ be a later step, with master points $v^{(m)},v^{(n)}$. Cuts are never removed, so $v^{(n)}\in\mathrm{Sat}(C)$ gives $j$ with $v^{(n)}_j>w^{(m)}_j$. Since $v^{(n)}_j\le M_j$, the coordinate $w^{(m)}_j$ is not clipped, so $v^{(n)}_j>v^{(m)}_j+\tau\ge v^{(m)}_j+\tau_0$. Here the strict cuts and the constraint $v\in B$ are used. Hence the cells $\lfloor v^{(m)}/\tau_0\rfloor\in\set{0,\dots,\lfloor M_{\max}/\tau_0\rfloor}^\iota$ of the NO steps are pairwise different: a coordinate that grows by at least $\tau_0$ moves to a higher cell. By the pigeonhole principle there are at most $(\lfloor 2LM_{\max}/(\eta\OPT)\rfloor+1)^{\abs\iota}$ NO steps. Together with step~0 and the YES steps this stays below the bound, a contradiction.
\end{proof}

\paragraph{The implemented algorithm as a refinement.}
The program refines the abstract method, and each refinement adds only constraints valid on $H$ or extra queries; none of them is part of the formal model (\cref{sec:lean}). The master is a MILP over $z\in B$ with year signs $\sigma_t\in\set{0,1}$ and sign rows $\alpha_t\le\mathbf M^\alpha_t\sigma_t$, $\beta_t\le\mathbf M^\beta_t(1-\sigma_t)$. Its dual bound is a bound over one-signed points only, which is what the abstract master must provide, since the representatives are one-signed (\cref{thm:onesigned}). Dominance rows and the cover cuts of \cref{sec:conflicts} hold on all of $H$ (\cref{prop:cover,thm:certificate}), so they remove no representative. For $r>1$ the objective $\norm{c\circ(\alpha+\beta)}_r$ is replaced by tangents, a lower bound. The gap $1/4$ is an assumption on the solver; $\UB$ never depends on it. A $\delta$-query of the box $v+\delta\e$, $\delta=\eta\UB/(4L)$, absorbs floating-point coincidences: a YES updates $\UB$ and a NO is a valid cut by upward closedness. The bisection over boxes $v+s\e$, $s\ge\tau$ \citep[p.~479]{Tuy2000}, starts at the model's query, and a NO at $s\ge\tau$ gives a cut point $\ge w$, so the separation argument holds with the same $\tau_0$. A NO on a subset $Y$ of the years gives the polyblock cut at the point equal to $\hat z$ on $Y$ and to $\mathbf M$ elsewhere, which lies in $B\setminus H$ by the observation on deleting years in the proof of \cref{thm:reduction}. Corners are rounded so that each queried box contains the master box. $\UB$ is the box cost of a box that passed the query. The final rational $Q$ is built from the best box (\cref{prop:rational-Q}) and verified exactly in rational arithmetic, and $C_r(Q)^r$ is rational for integer $r$. $\LB$ rests on the MILP solver up to its tolerances.

\begin{proposition}[closedness]\label{prop:closed}
\leavevmode
\begin{enumerate}[label=(\alph*)]
\item The set of half-widths $z=(\alpha,\beta)$ passing $\BOX$ is closed.
\item If $\BOX(0,0)$ fails and $c>0$, then $\OPT>0$, and the bound of \cref{thm:algo}(b) holds for every $\eta>0$.
\end{enumerate}
\end{proposition}
\begin{proof}
(a) For fixed $S$, the families $W$ at which $S$ is weakly reachable form a closed set: the witness of each $t\in S$ can be scaled into the standard simplex, the pairs $(\xi,W)$ with $\ip\xi{W_s-W_t}\ge0$ for $s\notin S$ form a closed set, and the projection along a compact factor is closed. The bounds of the query do not depend on $z$, and $z$ passes if and only if some finite family $\mathcal A$ of sets, all weakly reachable at $W(\alpha,\beta)$, carries a cone vector within the bounds. So the feasible set is a finite union over such $\mathcal A$ of finite intersections of preimages of closed sets under the continuous map $z\mapsto W(\alpha,\beta)$. (b) The complement of the feasible set is open and contains $0$, so some cube $[0,r)^\iota$ is infeasible. Every feasible $z$ has a coordinate $\ge r$, hence $f(z)\ge r\min_{t,i}c^i_t>0$.
\end{proof}
If $\BOX(0,0)$ holds, step~0 stops with $\UB=\LB=f(0)$. Step~0 is needed: with $\LB=0$ the relative stopping test never holds when $\OPT=0$ is not attained at once.

\begin{remark}[bound, stability and open questions]
The step bound in \cref{thm:algo}(b) is exponential in $2dT$. It guarantees termination, not speed. With a YES/NO oracle only, exponentially many queries cannot be avoided in general, and polyblock methods are reliable only in small dimension \citep{TuyAlKhayyalThach2005}. The cover cuts describe the curved boundary of the feasible set, which polyblock cuts approximate slowly. Since the feasible set is closed (\cref{prop:closed}), essentially optimal solutions \citep{Tuy2005} are not needed. The restriction $\mathbf M^\alpha<1$ means that no price is moved down by $100\%$ or more; it is irrelevant on data where the corrections are below $25\%$. The complexity of the problem for $d=2$, where the oracle is polynomial, remains open, as does the question whether polynomially many oracle calls with certificates suffice.
\end{remark}
