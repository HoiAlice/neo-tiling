\section{The least $\ell_p$ perturbation: an exact algorithm}\label{sec:lp}

\subsection{Problem}\src{l1-perturbation.tex \S1}
Let $g\colon\R\to\R_{++}$ be increasing with $g(0)=1$; the deviation of $q$ from $p$ is $\dev(q,p):=g^{-1}(q/p)$, coordinatewise. Two instances: $g(z)=e^z$ (logarithmic, $\dev=\ln q-\ln p$) and $g(z)=1+z$ (relative, $\dev=q/p-1$, the price error of \citet{Varian1985,DemuynckRehbeck2023}). Weights $c^i_t>0$, $1\le p\le\infty$,
\[
C_p(Q):=\norm{c\circ\dev(Q,P)}_p,\qquad\OPT_p:=\inf\set{C_p(Q)\mid y\text{ weakly solvable at }Q}.
\]
To half-widths $\alpha,\beta\in\Rp[d\times T]$ ($\alpha<1$ in the relative metric) associate the box with corners $a_t:=p_t\circ g(-\alpha_t)$, $b_t:=p_t\circ g(\beta_t)$; $\BOX(\alpha,\beta)$ is the box-oracle query of \cref{sec:corners} on $W(\alpha,\beta)=(a,b)$.

\subsection{One-signedness}\src{l1-perturbation.tex \S3}
\begin{theorem}[one-signedness]\label{thm:onesigned}
There is an optimal $Q$ in which every year moves in one direction: for each $t$ either $q_t\le p_t$ or $q_t\ge p_t$. \hfill\lean{exists_oneSigned_le}, \lean{exists_lt_of_mixed_lp}

\subsection{Covers and the certificate of infeasibility}\src{l1-perturbation.tex \S4}
\begin{definition}\todo{translate: cover $(t,A)$ present in $W$.}\end{definition}
\begin{proposition}[cover]\label{prop:cover}
Every weakly reachable set containing $t$ meets every $A$ whose convex hull has a point strictly below $w_t$; in particular dominance $w_s<w_t$ forces $s\in S$ whenever $t\in S$. \hfill\lean{meets_of_conv_lt}, \lean{le_of_lt}
\end{proposition}
\begin{theorem}[certificate]\label{thm:certificate}
Let the oracle answer NO on $W$ with a Farkas vector $\lambda$ and a finite set $\mathcal C$ of covers present in $W$ such that $\lambda(S)\ge0$ for every $S$ respecting $\mathcal C$ and $\ip\lambda{\cdot}<0$ on the bounds. Then the answer is NO on every box in which all covers of $\mathcal C$ are present. \hfill\lean{not_boxCone_of_certificate}
\end{theorem}
