\section{Least price perturbation: corners, breakpoints and the least radius}\label{sec:corners}

Prices are $(d+1)$-tuples of positive numbers known up to a common factor; normalizing by price $0$ gives $p\in\Rpp[d]$. Data: $P:=\set{p_t\in\Rpp[d]}_{t\in[T]}$ and $y\in\Rp[T]$. Each price $i=0,\dots,d$ in each period $t$ carries a penalty $\kappa^i_t>0$; a larger penalty means a more precisely measured price. $\exp$ and $\ln$ of vectors are coordinatewise. Indices of spectra may range over any finite set of points of $\Rp[d]$.

\subsection{The metric}\src{price-perturbation.tex \S2}
\[
d_{\kappa_t}(q,p):=\min_{s\in\R}\max\Bigl(\kappa^0_t\abs s,\ \max_{i\in[d]}\kappa^i_t\abs{u_i+s}\Bigr),\qquad u:=\ln q-\ln p .
\]
\begin{lemma}[shape of the ball]\label{lem:shape}
The ball $\set{q\mid d_{\kappa_t}(q,p_t)\le\rho}$ is, in log coordinates, the zonotope \enquote{box $\prod_i[-\rho/\kappa^i_t,\rho/\kappa^i_t]$ plus the diagonal segment of half-length $\rho/\kappa^0_t$}. Its bounding box has half-widths $a^i_t=\rho v^i_t$, $v^i_t:=1/\kappa^i_t+1/\kappa^0_t$; the corners $\pm a_t$ and the path $\theta\mapsto\theta v_t$ lie in the ball. \hfill\lean{wdist_le_iff}, \lean{mem_wbox_of_wdist}, \lean{wdist_wpath_le}
\end{lemma}

\begin{theorem}[corners]\label{thm:corners}
Let $d\ge2$, $y\in\Rp[T]$, $\rho\ge0$, $a_t:=p_t\circ e^{-\rho v_t}$, $b_t:=p_t\circ e^{\rho v_t}$, $W:=(a,b)$. The following are equivalent:
\begin{enumerate}[label=(\Alph*)]
\item there is $Q\in\Rpp[d\times T]$ with $d_{\kappa_t}(q_t,p_t)\le\rho$ for all $t$ at which $y$ is weakly solvable;
\item there are $\theta_t\in[-\rho,\rho]$ such that $y$ is weakly solvable at $(p_t\circ e^{\theta_tv_t})_{t\in[T]}$;
\item there is $w\in\cone\set{\ind S\mid S\in\wSp(W)}$ with $w(a_t)\ge y_t\ge w(b_t)$ for all $t\in[T]$.
\end{enumerate}
\hfill\lean{wfeasible_iff_corners}, \lean{wfeasible_iff_path}; general box form \lean{box_iff}, \lean{path_of_corners}
\end{theorem}

\paragraph{The box oracle.} Condition (C) is one query to the \emph{box oracle}: given points $W$ indexed by $R$ and bounds $\mathrm{lo},\mathrm{hi}\in[-\infty,\infty]^R$, decide whether some $w\in\cone\set{\ind S\mid S\in\wSp(W)}$ satisfies $\mathrm{lo}\le w\le\mathrm{hi}$. With $\mathrm{lo}=\mathrm{hi}=y$ and $W=P$ this is weak solvability (\cref{thm:closure}); with $W=(a,b)$, $\mathrm{lo}=(y,0)$, $\mathrm{hi}=(\infty,y)$ it is (C). Thus \enquote{is there $Q$ in the box $[a,b]$ at which $y$ is weakly solvable} is \emph{one} call of the same algorithm on $2T$ points with inequalities instead of equalities.

\subsection{The least radius}\src{price-perturbation.tex \S5}
\begin{theorem}[least radius]\label{thm:least}
The set $F$ of feasible radii is $[\rho^*,\infty)$ or empty, and $\rho^*=0$ or $\rho^*$ is a breakpoint: a zero of a nonzero exponential sum $\sum_jC_je^{\lambda_j\rho}$ given by a $(d+1)$-minor of the homogenized corners. \hfill\lean{least_wradius_breakpoint}
\end{theorem}

\begin{proposition}[equal penalties]\label{prop:rational}
If $v^i_t=c$ for all $i,t$, every $(d+1)$-minor of the vectors $(p_t,1)$, $(zp_t,1)$, $(e_i,0)$, $z:=e^{2c\rho}$, equals $z^{b-1}(D_0z+D_1)$; hence all breakpoints are rational in $z$ and $e^{2c\rho^*}\in\Q$ for rational data. \hfill\lean{least_radius_rational}
\end{proposition}

\begin{theorem}[complexity]\label{thm:complexity}
Let $d$ be fixed and the data $P,y,\kappa$ rational of total bit length $L$.
\begin{enumerate}[label=(\alph*)]
\item If the penalties are equal, $e^{2c\rho^*}$ is rational and $\rho^*$ is found exactly with $O(d\log(T+d))$ oracle calls and $O((2T+d)^{d+1})$ determinants of order $d+1$ on numbers of length $O(dL)$. The number of calls does not depend on $L$.
\item For arbitrary penalties there is $\bar\rho$ with $\log\bar\rho=\mathrm{poly}(L)$ such that $F\ne\varnothing$ iff $\bar\rho\in F$; bisection with safe rounding finds an interval of length $\varepsilon$ containing $\rho^*$ with $\mathrm{poly}(L)+O(\log(1/\varepsilon))$ oracle calls, all on rational data of length $\mathrm{poly}(L,\log(1/\varepsilon))$.
\end{enumerate}
For $d=2$ the oracle is a linear program of polynomial size (\cref{thm:lp}), so (a) is polynomial and (b) weakly polynomial.
\end{theorem}
