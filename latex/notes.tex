\section{Разрешимость}
\begin{definition}[Неоклассические Функции]
    Пусть $n\in\mathbb{N}$
    Будем называть функцию $f:\mathbb{R}_+^n\to\mathbb{R}_+$ \textit{неоклассической}, если она \textit{непрерывна}, \textit{вогнута} и однородна степени~1.\\
    Будем обозначать класс неоклассических функций $n$ переменных как $\Phi_n$.
\end{definition}

\begin{definition}[CES Функции]
    Пусть $n\in\mathbb{N}$, $J\subset[n]$, $\epsilon\in[-1,0)\cup(0,+\infty)$. Тогда отображение $A:J\to\mathbb{R}_{++}$ задает параметрическое семейство неоклассических функций
    \[H(X) := \Bigl(\sum_{j\in J}\Bigl(\frac{X_{j}}{A_{j}}\Bigr)^{-\epsilon}\Bigr)^{-1/\epsilon},\]
    называемых функциями \textit{постоянной эластичности замещения} $\sigma := \frac{1}{1+\epsilon}$.
    При $\epsilon\to 0$ семейство сходится к функции Кобба--Дугласа ($\sigma=1$), при $\epsilon\to+\infty$ --- к леонтьевской $\min_{j\in J} X_j/A_j$ ($\sigma=0$), при $\epsilon=-1$ --- к линейной $\sum_{j\in J} X_j/A_j$ ($\sigma=\infty$). При $\epsilon<-1$ функция не вогнута и потому не является неоклассической. Обозначим класс CES функций $\Psi_n\subset\Phi_n$.
\end{definition}

\begin{definition}[Неоклассическая разрешимость]
    Пусть $\{\hat p_t \in \Rpp\}_{t\in[T]}$ --- некоторый конечный набор цен.
    Будем говорить, что вектор выпусков $y\in\mathbb{R}^{T}_{++}$ \textit{неоклассически разрешим} при этих ценах, если
    существует такая неоклассическая функция $h\in\Phi_{2}$ и такие окрестности
    $\{U_t\ni\hat p_t\}_{t\in[T]}$, что для каждого набора $\{p_t\in U_t\}_{t\in[T]}$
    существует полная абсолютно непрерывная мера $\mu$ на $\Rp$, что $\forall t\in[T]\Rightarrow\mu(A_t)=y_t$, где ${A_t=\{x\in\Rpp\mid h(p_t\circ x)<1\}}.$
\end{definition}

\begin{definition}[CES-разрешимость]
    Будем говорить, что вектор выпусков $y\in\mathbb{R}^{T}_{++}$ \textit{CES-разрешим} при ценах $\{\hat p_t\}_{t\in[T]}$, если в предыдущем определении функцию $h$ можно выбрать из класса $\Psi_{2}$. Поскольку $\Psi_{2}\subset\Phi_{2}$, из CES-разрешимости следует неоклассическая разрешимость.
\end{definition}

\begin{proposition}
    $\exists q\in conv\{p_{1},p_{3}\}:q\leq p_{2}\Rightarrow A_{2}\subset A_{1}\cup A_{3}$.
\end{proposition}

\begin{example}
    При ценах $\{\hat{p}_{1} = (4, 1), \hat{p}_{2} = (3, 3), \hat{p}_{3} = (1, 4)\}$ вектор выпусков $y=(1, 5, 3)$ не является неоклассически разрешимым (а значит и CES-разрешимым).
    Более того, для случая CES функций, критическое значение параметра $\epsilon_{crit}\approx -2$, а значит ромбический тайлинг будет иметь вид $\lambda$-гекса независимо от эластичности замещения. Вектору выпусков $y=(1, 5, 3)$ соответствует змейка $(0,\xi_{2},\xi_{2}+\xi_{3},\xi_{1}+\xi_{2}+\xi_{3})$ которая лежит внутри зоногона $Z_{3}$ но не содержится в $\lambda$-гексе. 
\end{example}

\begin{example}
    При ценах $\hat p_1=(1,9)$, $\hat p_2=(4,4)$, $\hat p_3=(9,1)$, $\hat p_4=(10,100000)$, $\hat p_5=(20000,20000)$, $\hat p_6=(100000,10)$
    ромбический тайлинг любой CES функции состоит из двух гексов, склеенных в вершине со спектром $\{1,2,3\}$, каждый из которых имеет вид $\lambda$ или $\gamma$. Нижний гекс флипается при $\epsilon\approx-0.50$, а верхний --- при $\epsilon\approx-0.42$, поэтому при проходе $\epsilon$ по области допустимых значений возможны всего три спектра $Sp_{\lambda\lambda}$, $Sp_{\gamma\lambda}$, $Sp_{\gamma\gamma}$, а четвёртый спектр $Sp_{\lambda\gamma}$ невозможен. Вектор выпусков $y=(5,9,6,3,1,4)$ лежит в конусе $C(Sp_{\lambda\gamma})$ и не лежит ни в одном из конусов $C(Sp_{\lambda\lambda})$, $C(Sp_{\gamma\lambda})$, $C(Sp_{\gamma\gamma})$, а значит, не является CES-разрешимым.
    В то же время для неоклассической функции ${h(x):=(x^{1}x^{2})^{0.15}(x^{1}+x^{2})^{0.7}}$ линии уровня $h(\hat{p}_{t}\circ x)=1$ образуют строгую проволочную диаграмму с невозможным для CES функций спектром $Sp_{\lambda\gamma}$, значит $y$ неоклассически разрешим.
\end{example}

\section{Конечность Пересечений}

\begin{proposition} Функция $h(x):=\sqrt{x^{1}x^{2}}\exp(\frac{1}{4}\sin\ln\frac{x^{1}}{x^{2}})$ является неоклассической.
\end{proposition}

\begin{proof}
    Сначала покажем, что эта функция непрерывна на $\Rp$. Непрерывность на $\Rpp$ выполнена в силу того, что $h$ представима как композиция непрерывных функций. Для того, чтобы показать непрерывность на $\partial\Rp$ заметим, что  для каждого $x\in\Rpp$ значение функции $h(x)\in[e^{\frac{-1}{4}}\sqrt{x^{1}x^{2}}, e^{\frac{+1}{4}}\sqrt{x^{1}x^{2}}]$ а значит также как и в случае функций Кобба-Дугласса, наша функция продолжается по непрерывности на $\partial\Rp$ тождественным нулем.

    Теперь покажем, что эта функция вонута. Введем $u:=\ln\frac{x^{1}}{x^{2}}$, тогда гессиан имеет вид:
    \[
        \nabla^{2}h(x)
        = -\,\frac{h(x)\,\bigl(1+\sin u\bigr)\bigl(3+\sin u\bigr)}{16\,(x^{1})^{2}(x^{2})^{2}}
        \begin{pmatrix}
            (x^{2})^{2} & -x^{1}x^{2}\\[2pt]
            -x^{1}x^{2} & (x^{1})^{2}
        \end{pmatrix}.
    \]
    Множитель перед матрицей неположителен, а сама матрица имеет собственные числа $0$ и $(x^{1})^{2}+(x^{2})^{2}$, а значит сам гессиан отрицательно полуопределен и функция $h$ вогнута.
\end{proof}

\begin{example}
    Пусть $h(x)=\sqrt{x^{1}x^{2}}\exp(\frac{1}{4}\sin\ln\frac{x^{1}}{x^{2}})$ и $p\in\Rpp$, тогда если выполнено неравенство $|\ln p^{1} + \ln p^{2}|<|\sin\frac{\ln p^{1} - \ln p^{2}}{2}|$ то кривые $h(x)=1$ и $h(p\circ x)=1$ имеют счетное пересечение.
\end{example}

\begin{proof}
    Введем замену координат $\theta:= \ln x^{1} + \ln x^{2}$ и $\eta := \ln x^{1} - \ln x^{2}$, в этих координатах кривая $h(x)=1$ имеет простой параметрический вид $\theta =  -\frac{\sin\eta}{2}$ и определена для любого $\eta$.

    Помимо равенства $h(x)=1$ для точки пересечения также требуется $h(x)=h(p\circ x)$.
    Полагая, что $h(x)>0$ в координатах $(\theta,\eta)$ это уравнение представимо в следующем виде:
    \[
    2\ln\frac{h(p\circ x)}{h(x)}
    = (\ln p^{1} + \ln p^{2})
    + \sin\frac{\ln p^{1} - \ln p^{2}}{2}\,\cos\left(\eta + \frac{\ln p^{1} - \ln p^{2}}{2}\right) = 0.
    \]
    Это уравнение зависит от $\theta$ и имеет $0$ корней при $|\ln p^{1} + \ln p^{2}|>|\sin\frac{\ln p^{1} - \ln p^{2}}{2}|$, континуум решений в вырожденных случаях $p = (\exp(\pi z), \exp(-\pi z))$ для $z\in\mathbb{Z}$ и счетное кол-во в остальных случаях в том числе при $|\ln p^{1} + \ln p^{2}|<|\sin\frac{\ln p^{1} - \ln p^{2}}{2}|$. Для любого $\eta$ существует единственная $\theta$, которая задает точку на $h(x)=1$, что дает счетное кол-во пересечений. Более того, множество таких параметров $p$ открыто, а значит счетное кол-во пересечений достигается в \textit{общем положении}.
\end{proof}

\begin{definition}[Строго Положительная Неоклассическая Функция] Будем называть неоклассическую функцию $h\in\Phi_{n}$ \textit{строго положительной} если $\forall p\in\mathbb{R}^{n}_{+}/\set{0}\Rightarrow h(p)>0$. Класс строго положительных функций $n$ переменных будем обозначать как $\hat{\Phi}_{n}$.
\end{definition}

\begin{example} Пусть $J$ - конечное множество индексов, $A:J\to\Rpp$ тогда функция\\
    $h(p):=\min\limits_{j\in J}\set{A^{1}_{j}p^{1} + A^{2}_{j}p^{2}}$
    является строго положительной неоклассической функцией.
\end{example}

\begin{definition}[Конус Мордуховича~\cite{Mordukhovich2006}]
Пусть $A\subset\R^n$ замкнуто, $\bar x\in A$. \\
\emph{регулярный нормальный конус Фреше}:
\[\widehat N_A(\bar x):=\Bigl\{v\in\R^n\Bigm|
\limsup_{x\to\bar x,x\in A,x\ne\bar x}
\frac{\ip{v}{x-\bar x}}{\abs{x-\bar x}}\le 0\Bigr\},\]
\emph{предельный нормальный конус Мордуховича}:
\[N_A(\bar x):=\bigl\{v\in\R^n\bigm|\exists x_k\to\bar x,\ x_k\in A,\
v_k\to v,\ v_k\in\widehat N_A(x_k)\bigr\}.\]
Для выпуклого $A$ оба конуса совпадают с классическим определением нормального конуса.
\end{definition}

\begin{definition}[Трансверсальность~\cite{Kruger2006,Ioffe2017}]
Пусть $A,B\subset\R^n$ замкнуты, $x_{0}\in A\cap B$. Пара $\{A,B\}$
\emph{трансверсальна} в точке $x_{0}$, если
$N_A(x_{0})\cap\bigl(-N_B(x_{0})\bigr)=\{0\}.$
Набор $\{A_1,\dots,A_k\}$ трансверсален в $x_{0}\in\bigcap_i A_i$, если из
$v_1+\dots+v_k=0$, $v_i\in N_{A_i}(x_{0})$, следует $v_1=\dots=v_k=0$.
\end{definition}

\begin{theorem} Пусть $h,g\in\hat{\Phi}_{2},x_{0}\in\Rpp:h(\hat{x})=g(\hat{x})=1$. Линии уровня $h(x)=1$ и $g(x)=1$ пересекаются в $x_{0}$ трансверсально, если и только если $\partial h(x_{0})\cap\partial g(x_{0}) = \varnothing.$
Более того, точка $x_{0}\in\partial\Rpp$ не может быть трансверсальным пересечением этих кривых.
\end{theorem}

\begin{proof}
Строго положительная неоклассическая $h$ строго возрастает по каждой переменной (из супераддитивности $h(x+d)\ge h(x)+h(d)$), поэтому линия уровня $h(x)=1$ есть график выпуклой, непрерывной и строго убывающей функции $\varphi$ на отрезке $[0,b]$, $b=1/h(1,0)$, с $\varphi(b)=0$, а точка $x_0=(t_0,\varphi(t_0))$ внутренняя: $0<t_0<b$. Аналогично линия $g(x)=1$ --- график выпуклой $\psi$, $\psi(t_0)=\varphi(t_0)$. Супердифференциал $\partial h(x_0)$ отождествляется с множеством опорных наклонов $\partial\varphi(t_0)=[\varphi'_-(t_0),\varphi'_+(t_0)]$: подстановка $x=\tau x_0$ в опорное неравенство даёт тождество Эйлера $\ip{\xi}{x_0}=1$, подстановка $x=x_0+(0,1)$ даёт $\xi_2\ge h(0,1)>0$, а подстановка точек $(t,\varphi(t))$ линии уровня --- включение $-\xi_1/\xi_2\in\partial\varphi(t_0)$. Обратно, для $s\in\partial\varphi(t_0)$ вектор $\xi=\lambda(-s,1)$ с $\lambda=(\varphi(t_0)-s\,t_0)^{-1}>0$ удовлетворяет $\ip{\xi}{x_0}=1$ и $h(x)\le\ip{\xi}{x}$ на $\Rp$, поскольку для $x\ne0$ точка $x/h(x)$ лежит на линии $h(x)=1$, где опорное неравенство даёт $\ip{\xi}{x/h(x)}\ge1$. Отображение $s\mapsto\lambda(-s,1)$ обратимо и зависит только от $x_0$, поэтому условие $\partial h(x_0)\cap\partial g(x_0)=\varnothing$ равносильно $\partial\varphi(t_0)\cap\partial\psi(t_0)=\varnothing$.

Покажем, что предельный нормальный конус $N_h(x_0)$ линии уровня $h(x)=1$ в точке $x_0=(t_0,\varphi(t_0))$ состоит из клина $W=\set{\lambda(s,-1)\mid\lambda\ge0,\ s\in\partial\varphi(t_0)}$ и двух прямых, натянутых на его крайние лучи:
\begin{equation}\label{eq:cone}
N_h(x_0)=W\cup\ell_-\cup\ell_+,\quad
\ell_-=\{\mu(\varphi'_-(t_0),-1)\mid\mu\in\R\},\quad
\ell_+=\{\mu(\varphi'_+(t_0),-1)\mid\mu\in\R\}.
\end{equation}
Пусть $v$ --- нормаль Фреше к графику в точке $x_1=(t_1,\varphi(t_1))$. Подставляя в неравенство $\ip{v}{x-x_1}\le o(|x-x_1|)$ точки графика $x=(t,\varphi(t))$ и деля на $t-t_1$, в пределе при $t\downarrow t_1$ и при $t\uparrow t_1$ получаем
\begin{equation}\label{eq:frechet}
v_1+v_2\,\varphi'_+(t_1)\le0\le v_1+v_2\,\varphi'_-(t_1).
\end{equation}
Обратно, клин $W$ состоит из нормалей Фреше в силу опорного неравенства $\varphi(t)\ge\varphi(t_0)+s(t-t_0)$, а в точке дифференцируемости $\varphi$ нормалью Фреше является и вектор $(-\varphi'(t_1),1)$, так что там нормалями служит вся прямая $\R(\varphi'(t_1),-1)$. Кроме того, по лемме о трёх хордах $\varphi'_\pm(t)\to\varphi'_+(t_0)$ при $t\downarrow t_0$ и $\varphi'_\pm(t)\to\varphi'_-(t_0)$ при $t\uparrow t_0$. Включение $\subset$ в~\eqref{eq:cone}: если $v=\lim v_n$, где $v_n$ --- нормали Фреше в точках $(t_n,\varphi(t_n))$, $t_n\to t_0$, то, перейдя к подпоследовательности, можно считать, что все $t_n>t_0$, все $t_n<t_0$ или все $t_n=t_0$. В первых двух случаях предельный переход в~\eqref{eq:frechet} даёт $v_1+v_2\varphi'_\pm(t_0)=0$, то есть $v\in\ell_\pm$; в третьем $v$ удовлетворяет~\eqref{eq:frechet} в точке $t_0$, откуда $v\in W$ при $v_2\le0$, а при $v_2>0$ из $\varphi'_+(t_0)\le-v_1/v_2\le\varphi'_-(t_0)$ следует $\varphi'_-(t_0)=\varphi'_+(t_0)$ и $v\in\ell_+$. Включение $\supset$: клин $W$ уже разобран, вектор $(\varphi'_+(t_0),-1)$ лежит в $W$, а $(-\varphi'_+(t_0),1)$ есть предел нормалей Фреше $(-\varphi'(t_n),1)$ в точках дифференцируемости $t_n\downarrow t_0$; такие точки есть в любом интервале, поскольку неубывающая $\varphi'_+$ имеет не более чем счётное множество разрывов, а в точке её непрерывности из тех же пределов односторонних производных следует $\varphi'_-=\varphi'_+$. Для $\ell_-$ аналогично.

Наконец, $-W_\psi=\set{\lambda(-s,1)\mid\lambda\ge0,\ s\in\partial\psi(t_0)}$, а прямые симметричны, поэтому ненулевой вектор $v\in N_h(x_0)\cap(-N_g(x_0))$ не может лежать в обоих клиньях (знак второй координаты), а сравнение координат в остальных случаях даёт $\varphi'_\pm(t_0)\in\partial\psi(t_0)$, $\psi'_\pm(t_0)\in\partial\varphi(t_0)$ или $\varphi'_\pm(t_0)=\psi'_\pm(t_0)$ --- во всех случаях общую точку отрезков $\partial\varphi(t_0)$ и $\partial\psi(t_0)$. Обратно, если отрезки пересекаются, то конец одного лежит в другом, скажем $\varphi'_+(t_0)\in\partial\psi(t_0)$, и тогда $v=(-\varphi'_+(t_0),1)\in\ell_+$, $-v\in W_\psi$. Итак, трансверсальность равносильна $\partial\varphi(t_0)\cap\partial\psi(t_0)=\varnothing$, то есть $\partial h(x_0)\cap\partial g(x_0)=\varnothing$.

Пусть, наконец, $x_0$ лежит на оси, скажем $x_0=(b,0)$; случай другой оси симметричен. Линия $h(x)=1$ пересекает ось $x_2=0$ только в точке $(b,0)$, так как $h(t,0)=t\,h(1,0)$, поэтому вблизи $x_0$ она является дугой $(t,\varphi(t))$, $t\le b$, с концом в $x_0$, и в силу существования односторонней производной $\varphi'_-(b)$, конечной или равной $-\infty$, направления $u(x)=(x-x_0)/|x-x_0|$ при $x\to x_0$ вдоль дуги сходятся к единичному вектору $\tau$. Всякий $v$ с $\ip{v}{\tau}\le0$ есть нормаль Фреше в $x_0$: $\ip{v}{x-x_0}=|x-x_0|\ip{v}{u(x)}$, а $\ip{v}{u(x)}\to\ip{v}{\tau}\le0$. Значит, $N_h(x_0)$ содержит замкнутую полуплоскость $\set{\ip{v}{\tau}\le0}$, и точно так же $-N_g(x_0)$ содержит полуплоскость $\set{\ip{v}{\tau'}\ge0}$ для направления $\tau'$ дуги $g(x)=1$. Две замкнутые полуплоскости с граничными прямыми через нуль имеют общий ненулевой вектор (один из двух векторов, ортогональных $\tau$), поэтому $N_h(x_0)\cap(-N_g(x_0))\ne\set{0}$ и пересечение не трансверсально.
\end{proof}

\begin{theorem} Пусть $h\in\hat{\Phi}_{2}$, $U\subset\Rpp$ -- открытое непустое множество. Тогда существует непустое открытое подмножество $V\subset U$ что $\forall p\in V$ кривые $h(x)=1$ и $h(p\circ x) = 1$ имеют конечное кол-во пересечений, причем все эти пересечения происходят трансверсально.
\end{theorem}
