\section{Разрешимость}
\begin{definition}[Неоклассические Функции]
    Пусть $n\in\mathbb{N}$
    Будем называть функцию $f:\mathbb{R}_+^n\to\mathbb{R}_+$ \textit{неоклассической}, если она \textit{непрерывна}, \textit{вогнута} и однородна степени~1.\\
    Будем обозначать класс неоклассических функций $n$ переменных как $\Phi_n$.
\end{definition}

\begin{definition}
    Пусть $\{\hat p_t \in \mathbb{R}^{2}_{++}\}_{t\in[T]}$ --- некоторый конечный набор цен.
    Будем говорить, что вектор выпусков $y\in\mathbb{R}^{T}_{++}$ разрешим при этих ценах, если
    существует такая неоклассическая функция $h\in\Phi_{2}$ и такие окрестности
    $\{U_t\ni\hat p_t\}_{t\in[T]}$, что для каждого набора $\{p_t\in U_t\}_{t\in[T]}$
    существует полная неотрицательная абсолютно непрерывная мера $\mu$ на $\mathbb{R}^{2}_{+}$, что $\forall t\in[T]\Rightarrow\mu(A_t)=y_t$, где $A_t=\{x\in\mathbb{R}^{2}_{++}\mid h(p_t\circ x)<1\}.$
\end{definition}

\begin{proposition}
    $\exists q\in conv\{p_{1},p_{3}\}:q\leq p_{2}\Rightarrow A_{2}\subset A_{1}\cup A_{3}$.
\end{proposition}

\begin{example}
    При ценах $\{\hat{p}_{1} = (4, 1), \hat{p}_{2} = (3, 3), \hat{p}_{3} = (1, 4)\}$ вектор выпусков $y=(1, 5, 3)$ не разрешим.
    Более того, для случая $CES$ функций, критическая эластичность замещения $\rho_{crit}\approx -2$, а значит ромбический тайлинг будет иметь вид $\lambda$-гекса независимо от эластичности замещения. Вектору выпусков $y=(1, 5, 3)$ соответствует змейка $(0,\xi_{2},\xi_{2}+\xi_{3},\xi_{1}+\xi_{2}+\xi_{3})$ которая лежит внутри зоногона $Z_{3}$ но не содержится в $\lambda$-гексе. 
\end{example}

\begin{remark}
    Кажется, для $T=3$ не существует примера, для которого переход от $CES$ функций к неоклассическим дал бы больше разрешимых случаев. Для $T=6$ можно построить пример, основываясь на том что в ромбических тайлингах, построенных для $CES$ функций, флипы могут происходить только в определенном порядке. Для $\Phi_2$ этот порядок можно варьировать. \textbf{Необходимо добавить пример.}
\end{remark}

\begin{proposition} Функция $h(x):=\sqrt{x^{1}x^{2}}\exp(\frac{1}{4}\sin\ln\frac{x^{1}}{x^{2}})$ является неоклассической.
\end{proposition}

\begin{proof}
    Сначала покажем, что эта функция непрерывна на $\mathbb{R}^{2}_{+}$. Непрерывность на $\mathbb{R}^{2}_{++}$ выполнена в силу того, что $h$ представима как композиция непрерывных функций. Для того, чтобы показать непрерывность на $\partial\mathbb{R}^{2}_{+}$ заметим, что  для каждого $x\in\mathbb{R}^{2}_{++}$ значение функции $h(x)\in[e^{\frac{-1}{4}}\sqrt{x^{1}x^{2}}, e^{\frac{+1}{4}}\sqrt{x^{1}x^{2}}]$ а значит также как и в случае функций Кобба-Дугласса, наша функция продолжается по непрерывности на $\partial\mathbb{R}^{2}_{+}$ тождественным нулем.

    Теперь покажем, что эта функция вонута. Введем $u:=\ln\frac{x^{1}}{x^{2}}$, тогда гессиан имеет вид:
    \[
        \nabla^{2}h(x)
        = -\,\frac{h(x)\,\bigl(1+\sin u\bigr)\bigl(3+\sin u\bigr)}{16\,(x^{1})^{2}(x^{2})^{2}}
        \begin{pmatrix}
            (x^{2})^{2} & -x^{1}x^{2}\\[2pt]
            -x^{1}x^{2} & (x^{1})^{2}
        \end{pmatrix}.
    \]
    Множитель перед матрицей неположителен, а сама матрица имеет собственные числа $0$ и $(x^{1})^{2}+(x^{2})^{2}$, а значит сам гессиан отрицательно полуопределен и функция $h$ вогнута.
\end{proof}

\begin{example}
    Пусть $h(x)=\sqrt{x^{1}x^{2}}\exp(\frac{1}{4}\sin\ln\frac{x^{1}}{x^{2}})$ и $p\in\mathbb{R}^{2}_{++}$, тогда если выполнено неравенство $|\ln p^{1} + \ln p^{2}|<|\sin\frac{\ln p^{1} - \ln p^{2}}{2}|$ то кривые $h(x)=1$ и $h(p\circ x)=1$ имеют счетное пересечение.
\end{example}

\begin{proof}
    Введем замену координат $\theta:= \ln x^{1} + \ln x^{2}$ и $\eta := \ln x^{1} - \ln x^{2}$, в этих координатах кривая $h(x)=1$ имеет простой параметрический вид $\theta =  -\frac{\sin\eta}{2}$ и определена для любого $\eta$.

    Помимо равенства $h(x)=1$ для точки пересечения также требуется $h(x)=h(p\circ x)$.
    Полагая, что $h(x)>0$ в координатах $(\theta,\eta)$ это уравнение представимо в следующем виде:
    \[
    2\ln\frac{h(p\circ x)}{h(x)}
    = (\ln p^{1} + \ln p^{2})
    + \sin\frac{\ln p^{1} - \ln p^{2}}{2}\,\cos\left(\eta + \frac{\ln p^{1} - \ln p^{2}}{2}\right) = 0.
    \]
    Это уравнение зависит от $\theta$ и имеет $0$ корней при $|\ln p^{1} + \ln p^{2}|>|\sin\frac{\ln p^{1} - \ln p^{2}}{2}|$, континуум решений в вырожденных случаях $p = (\exp(\pi z), \exp(-\pi z))$ для $z\in\mathbb{Z}$ и счетное кол-во в остальных случаях в том числе при $|\ln p^{1} + \ln p^{2}|<|\sin\frac{\ln p^{1} - \ln p^{2}}{2}|$. Для любого $\eta$ существует единственная $\theta$, которая задает точку на $h(x)=1$, что дает счетное кол-во пересечений.
\end{proof}

\section{Конечность Пересечений}

\begin{proposition}
    Пусть $ H,h\in\Phi_{2}$ ненулевые неоклассические функции, что:
    \[
        h(p) = \inf\limits_{x\geq0}\set{\langle p,x\rangle \mid H(x)\geq 1},
        \qquad
        H(x) = \inf\limits_{p\geq0}\set{\langle x,p\rangle \mid h(p)\geq 1}.
    \]
    Тогда $\forall a,b>0$ верна следующая эквивалентность:
    \[\forall p\in\mathbb{R}^{2}_{+}\Rightarrow h(p)\geq ap^{1} + bp^{2} \iff \forall x\in\mathbb{R}^{2}_{+}\Rightarrow H(x)\leq\min\set{\frac{x^{1}}{a},\frac{x^{2}}{b}}.\]
\end{proposition}

\begin{lemma}\label{lem:no-intersect}
    Пусть $a,b>0$ и $h\in\Phi_{2}:h(x)\geq ax^{1}+bx^{2}$. Также пусть $p^{1},p^{2}>1$, тогда кривые $h(x)=1$ и $h(p\circ x)=1$ не имеют пересечений.
\end{lemma}

\begin{proof}
    Предположим $x_{*}$ - точка пересечений кривых $h(x)=1$ и $h(p\circ x)=1$. Но тогда $1 = h(p\circ x_{*})\geq\min\set{p^{1},p^{2}}h(x_{*}) > h(x_{*}) = 1$ что приводит к противоречию $1>1$.
\end{proof}

\begin{lemma}\label{lem:finite-omega} Пусть $f:[0,1]\to[0,1]$ --- непрерывная, выпуклая, монотонно убывающая функция, причем $f(0)=1, f(1)=0$. Также пусть $\alpha>1>\beta>0$. Тогда для каждой окрестности $U\ni(\alpha,\beta)$ найдется точка $(\hat{\alpha},\hat{\beta})\in U$ что множество пересечений \\
$\hat{\Omega}:=\set{y\in [0,1]\,\middle|\,\hat{\alpha} f(y) = f(\hat{\beta}y)}$ конечно, а сами пересечения трансверсальны.
\end{lemma}

\begin{proposition}\label{prop:finite}
    Пусть $a,b>0$ и $h\in\Phi_{2}:h(x)\geq ax^{1}+bx^{2}$. Тогда для любой окрестности $U\subset\mathbb{R}^{2}_{++}$ существует подокрестность $V\subset U$ что для каждого $p\in V$ кривые $h(x)=1$ и $h(p\circ x)=1$ пересекаются конечное кол-во раз.
\end{proposition}

\begin{proof}
    Зафиксируем $U\subset\mathbb{R}^{2}_{++}, p\in U$ и рассмотрим случаи. Так как $U$ открыто то в нем можно выделить подокрестность $V\subset U:\forall p\in V\Rightarrow p^{1},p^{2}\neq 1$. В дальнейшем положим $U := V$.
    Тогда возможны два принципальных случая: $p^{1}, p^{2} > 1$ и $p^{1} > 1 > p^{2}$. В первом случае по лемме~\ref{lem:no-intersect} пересечений $h(x)=1$ и $h(p\circ x)=1$ не имеют. При анализе второго случая удобно будет рассмотреть неявную функцию $h(f(y), y) = 1$. Так как $h(x)\geq ax^{1} + bx^{2}$ для некоторых $a,b>0$, то $\exists\hat{x}\in\mathbb{R}^{2}_{+}:h(\hat{x}^{1},0)=h(0,\hat{x}^{2})=1$. Функция $f:[0,\hat{x}^{2}]\to[0,\hat{x}^{1}]$ оказывается монотонно убывающей, непрерывной и выпуклой. Кривая $h(p\circ x)=1$ также может быть задана функцией $y\mapsto\frac{1}{p^1}f(p^2 y)$. Тогда условие пересечение выражается как равенство $y\in[0,\hat{x}^{2}], p^{1}f(y) = f(p^{2}y)$. По лемме~\ref{lem:finite-omega} существуют $\hat{p}\in U$ для которых функции $\hat{p}^{1}f(y)$ и $f(\hat{p}^{2}y)$ на отрезке $[0,\hat{x}^{2}]$ пересекаются конечное кол-во раз и все пересечения трансверсальны. Более того, $\hat{p}^1>1>\hat{p}^2$, а значит все пересечения лежат на отрезке $[2\varepsilon,\hat{x}^{2}-2\varepsilon]$ для некоторого $\varepsilon>0$. Выпуклая на $[0,\hat{x}^{2}]$ функция $f$ липшицева на $[\varepsilon,\hat{x}^{2}-\varepsilon]$.
\end{proof}
