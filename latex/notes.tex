\section{Разрешимость}

Всюду $d\in\N$, $d\ge2$. Для $p,x\in\R^d$ через $p\circ x:=(p^1x^1,\dots,p^dx^d)$ обозначим покоординатное произведение, через $\mathbf e:=(1,\dots,1)\in\R^d$~--- единичный вектор, через $\mathbf 1_S\in\R^T$~--- индикатор множества $S\subset[T]$. Неравенства между векторами покоординатные.

\begin{definition}[Неоклассические Функции]\label{def:neoclassical}
    Будем называть функцию $f:\Rp[d]\to\R_{+}$ \textit{неоклассической}, если она \textit{непрерывна}, \textit{вогнута} и однородна степени~1.\\
    Будем обозначать класс неоклассических функций $d$ переменных как $\Phi_d$.
\end{definition}

\begin{definition}[Неоклассическая разрешимость]\label{def:neo-solvability}
    Пусть $\set{\hat p_t\in\Rpp[d]}_{t\in[T]}$ --- некоторый конечный набор цен.
    Будем говорить, что вектор выпусков $y\in\Rp[T]$ \textit{неоклассически разрешим} при этих ценах, если
    существует такая неоклассическая функция $h\in\Phi_{d}$ и такие окрестности
    $\set{U_t\ni\hat p_t}_{t\in[T]}$, $U_t\subset\Rpp[d]$, что для каждого набора $\set{p_t\in U_t}_{t\in[T]}$
    существует абсолютно непрерывная мера $\mu$ на $\Rp[d]$, что $\forall t\in[T]\Rightarrow\mu(A_t)=y_t$, где ${A_t=\set{x\in\Rpp[d]\mid h(p_t\circ x)<1}}.$
\end{definition}

\section{Спектр}

\begin{definition}[Спектр]\label{def:spectrum}
    Пусть $h\in\Phi_d$ и $P:=\set{p_t\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Для $S\subset[T]$ назовём \textit{камерой} множество
    \[
        R_S:=\set{x\in\Rpp[d]\mid h(p_t\circ x)<1\ \text{при } t\in S,\ h(p_s\circ x)>1\ \text{при } s\notin S}.
    \]
    \textit{Спектром} пары $(h,P)$ назовём семейство $\Sp(h,P):=\set{S\subset[T]\mid R_S\neq\varnothing}$.
    Камеры открыты по непрерывности $h$ и попарно не пересекаются. Точка $x\in\Rpp[d]$ вне кривых $h(p_t\circ x)=1$ лежит в камере $R_{S(x)}$, $S(x):=\set{t\mid h(p_t\circ x)<1}$, поэтому $A_t:=\set{x\in\Rpp[d]\mid h(p_t\circ x)<1}$ отличается от $\bigsqcup_{S\ni t}R_S$ лишь точками кривых.
\end{definition}

\begin{lemma}\label{lem:null}
    Пусть $h\in\Phi_d$ и $p\in\Rpp[d]$. Тогда кривая $\Gamma:=\set{x\in\Rpp[d]\mid h(p\circ x)=1}$ имеет меру нуль.
\end{lemma}

\begin{proof}
    По однородности при $c>0$ множество $\set{x\in\Rpp[d]\mid h(p\circ x)=c}$ равно $c\Gamma$ и имеет меру $c^d\abs\Gamma$. Эти множества при разных $c$ попарно не пересекаются, поэтому положительную меру могут иметь лишь счётное их число. Значит, $\abs\Gamma=0$.
\end{proof}

\begin{lemma}\label{lem:signature}
    Рассмотрим неоклассическую функцию $h\in\Phi_{d}$ и набор цен $P:=\set{p_t\in\Rpp[d]}_{t\in[T]}$. Тогда вектор выпусков $y\in\Rp[T]$ принадлежит конусу $\cone\set{\mathbf 1_S\mid S\in\Sp(h,P)}$, если и только если существует абсолютно непрерывная мера $\mu$ на $\Rp[d]$, для которой при всех $t\in[T]$ выполнено интегральное условие $\mu(A_t)=y_t$, где $A_t:=\set{x\in\Rpp[d]\mid h(p_t\circ x)<1}$.
\end{lemma}

\begin{proof}
    По лемме~\ref{lem:null} кривые имеют меру нуль, поэтому $\mu(A_t)=\sum_{S\ni t}\mu(R_S)$, и условия $\mu(A_t)=y_t$ означают $y=\sum_S\mu(R_S)\mathbf 1_S$, где $\mu(R_S)=0$ при $S\notin\Sp(h,P)$. Обратно, при $S\in\Sp(h,P)$ камера $R_S$~--- непустое открытое множество, и равномерная вероятностная мера на шаре в $R_S$ реализует вектор $\mathbf 1_S$, а суммы и неотрицательные кратные мер реализуют суммы и кратные векторов.
\end{proof}

\begin{lemma}\label{lem:stability}
    Пусть $h\in\Phi_d$, $P\in\Rpp[d\times T]$. Тогда существует такая окрестность $V\subset\Rpp[d\times T]$ набора $P$, что для каждого набора цен $Q\in V$ выполнено $\Sp(h,Q)\supseteq\Sp(h,P)$.
\end{lemma}

\begin{proof}
    Пусть $S\in\Sp(h,P)$ и $x\in R_S$. Строгие неравенства $h(p_t\circ x)\gtrless1$ сохраняются при малом изменении цен по непрерывности $h$, поэтому $x$ лежит в камере со спектром $S$ и для $Q$, то есть $S\in\Sp(h,Q)$. Множеств $S$ конечное число.
\end{proof}

\section{Достижимые спектры}

\begin{definition}[Достижимые спектры]\label{def:reachable}
    Пусть $P:=\set{p_{t}\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Будем говорить, что $S\subset[T]$~--- \textit{достижимый спектр} при ценах $P$, если для каждого $t\in S$ существует ненулевой вектор $\xi\in\Rp[d]$, для которого $\ip{\xi}{p_s-p_t}>0$ при всех $s\notin S$. Такой вектор назовём \textit{свидетелем} индекса $t$. Будем обозначать как $\Sp(P)$ семейство таких спектров.
\end{definition}

\begin{lemma}\label{lem:necessity}
    Пусть $h\in\Phi_{d}$ и $P:=\set{p_{t}\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Тогда $\Sp(h,P)\subseteq\Sp(P)$.
\end{lemma}

\begin{proof}
    Пусть $S\in\Sp(h,P)$, $x\in R_S$ и $t\in S$. Если $S=[T]$, свидетелем служит любой ненулевой $\xi\in\Rp[d]$. Иначе множество $C:=\conv\set{p_s\circ x}_{s\notin S}+\Rp[d]$ выпукло и замкнуто как сумма компакта и замкнутого конуса. На $C$ выполнено $h\ge1$: при $r\notin S$ имеем $h(p_r\circ x)\ge1$, и по монотонности $h$ и неравенству Йенсена
    \[
        h\Bigl(\sum_{r\notin S}w_r\,p_r\circ x+v\Bigr)\ge h\Bigl(\sum_{r\notin S}w_r\,p_r\circ x\Bigr)\ge\sum_{r\notin S}w_r\,h(p_r\circ x)\ge1
    \]
    при $w_r\ge0$, $\sum_rw_r=1$, $v\ge0$. Так как $h(p_t\circ x)<1$, точка $p_t\circ x$ не лежит в $C$. По теореме об отделимости найдутся линейная функция $f$ и число $u$, что $f(p_t\circ x)<u<f(c)$ при всех $c\in C$. Функция $f$ неотрицательна на $\Rp[d]$: иначе при $f(v)<0$, $v\ge0$, точка $c+\lambda v\in C$ с подходящим $\lambda\ge0$ дала бы $f(c+\lambda v)=u$. Положим $\xi^i:=f(e_i)x^i\ge0$. Тогда $\ip{\xi}{p}=f(p\circ x)$, и при $s\notin S$ 
    $\ip{\xi}{p_s-p_t}=f(p_s\circ x)-f(p_t\circ x)>0$,
    откуда также $\xi\ne0$.
\end{proof}

\section{Критерий разрешимости}

\begin{lemma}[отделимость]\label{lem:separation}
    Пусть $S\in\Sp(P)$ и $t\in S$. Тогда существует вектор $\xi\in\Rpp[d]$, для которого $\ip{\xi}{p_t}<1<\ip{\xi}{p_s}$ при всех $s\notin S$.
\end{lemma}

\begin{proof}
    Пусть $\xi_0$~--- свидетель индекса $t$. При достаточно малом $\delta>0$ вектор $\xi_1:=\xi_0+\delta\mathbf e\in\Rpp[d]$ всё ещё удовлетворяет $\ip{\xi_1}{p_s-p_t}>0$ при всех $s\notin S$, так как таких $s$ конечное число. Положим $a:=\ip{\xi_1}{p_t}>0$ и выберем $c$ с $a<c<\min_{s\notin S}\ip{\xi_1}{p_s}$, а при $S=[T]$ положим $c:=a+1$. Тогда $\xi:=\xi_1/c$ искомый.
\end{proof}

\begin{lemma}[склейка]\label{lem:patch}
    Пусть $P:=\set{p_{t}\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Тогда существует кусочно-линейная $h\in\Phi_d$, для которой $\Sp(P)\subseteq\Sp(h,P)$.
\end{lemma}

\begin{proof}
    Для каждой пары $(S,t)$ с $t\in S\in\Sp(P)$ возьмём по лемме~\ref{lem:separation} вектор $\eta_{S,t}\in\Rpp[d]$ с $\ip{\eta_{S,t}}{p_t}<1$ и $\ip{\eta_{S,t}}{p_s}>1$ при $s\notin S$, и добавим к ним вектор $\eta_0:=\mathbf e$. Каждому $S\in\Sp(P)$ дадим свой показатель $E_S\in\N$, $E_S\ge1$, а вектору $\eta_0$~--- показатель $0$. Выберем $M\ge1$ так, чтобы $M\min(\eta^1p^1_s,\ \eta^2p^2_s)>1$ для всех выбранных векторов $\eta$ и всех $s\in[T]$, и для $z\in\Z$ положим $x_z:=(M^z,M^{-z},1,\dots,1)$. Тогда при $z\ne0$
    \[
        \ip{\eta}{p\circ x_z}\ \ge\ \eta^1p^1M^z+\eta^2p^2M^{-z}\ \ge\ M\min(\eta^1p^1,\ \eta^2p^2)>1,
    \]
    поскольку одно из чисел $M^z$, $M^{-z}$ не меньше $M$, а оба слагаемых положительны. Положим $\tilde\eta_{S,t}:=\eta_{S,t}\circ x_{-E_S}$, $\tilde\eta_0:=\eta_0$ и
    \[
        h(x):=\min\bigl(\ip{\tilde\eta_0}{x},\ \min_{t\in S\in\Sp(P)}\ip{\tilde\eta_{S,t}}{x}\bigr)\in\Phi_d.
    \]
    Так как $\ip{\eta\circ x_{-a}}{p\circ x_b}=\ip{\eta}{p\circ x_{b-a}}$, в точке $x_{E_S}$ векторы с показателем $E_S$ дают значения $\ip{\eta_{S,t}}{p}$, а все остальные~--- значения $\ip{\eta}{p\circ x_z}$ с $z\ne0$, большие единицы. Поэтому для $t\in S$ имеем $h(p_t\circ x_{E_S})\le\ip{\eta_{S,t}}{p_t}<1$, а для $s\notin S$~--- $h(p_s\circ x_{E_S})>1$. Значит, $x_{E_S}\in R_S$ и $S\in\Sp(h,P)$.
\end{proof}

\begin{theorem}\label{thm:spectrum}
    Пусть $P:=\set{p_{t}\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Тогда существует $h\in\Phi_d$, для которой $\Sp(h,P)=\Sp(P)$.
\end{theorem}

\begin{proof}
    Включение $\Sp(P)\subseteq\Sp(h,P)$ даёт лемма~\ref{lem:patch}, обратное~--- лемма~\ref{lem:necessity}.
\end{proof}

\begin{theorem}\label{thm:solvability}
    Пусть $P:=\set{p_{t}\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Тогда вектор выпусков $y\in\Rp[T]$ неоклассически разрешим при ценах $P$ тогда и только тогда, когда $y\in\cone\set{\mathbf 1_S\mid S\in\Sp(P)}$.
\end{theorem}

\begin{proof}
    Пусть $y$ разрешим с функцией $h\in\Phi_d$. Взяв в определении~\ref{def:neo-solvability} сами цены $P$, по лемме~\ref{lem:signature} получаем $y\in\cone\set{\mathbf 1_S\mid S\in\Sp(h,P)}$, а $\Sp(h,P)\subseteq\Sp(P)$ по лемме~\ref{lem:necessity}. Обратно, возьмём $h$ из теоремы~\ref{thm:spectrum}. По лемме~\ref{lem:stability} $\Sp(h,Q)\supseteq\Sp(h,P)=\Sp(P)$ для всех $Q$ из некоторой окрестности $P$, и по лемме~\ref{lem:signature} при каждом таком $Q$ существует абсолютно непрерывная мера $\mu$ с $\mu(A_t)=y_t$.
\end{proof}

\begin{remark}
    Функция $h$ из теоремы~\ref{thm:spectrum} не зависит от $y$. Тем самым каждый разрешимый вектор выпусков реализуется одной и той же функцией $h$, и от $y$ зависит только мера. Никаких условий общего положения на набор цен не требуется: совпадающие координаты и даже совпадающие цены допустимы.
\end{remark}

\paragraph{Алгоритм.} Теорема~\ref{thm:solvability} даёт конечную процедуру проверки неоклассической разрешимости $y$ при ценах $P$.
\begin{enumerate}
    \item \emph{Построение $\Sp(P)$.} Перебираем $S\subset[T]$ и для каждого $t\in S$ ищем свидетеля, то есть решаем задачу линейного программирования из замечания после определения~\ref{def:reachable}. Оставляем те $S$, у которых свидетели есть у всех $t\in S$. Это $2^T$ кандидатов и не более $T$ задач линейного программирования на каждый.
    \item \emph{Проверка принадлежности конусу.} Решаем задачу линейного программирования: найти $m_S\ge0$, $S\in\Sp(P)$, с $\sum_Sm_S\mathbf 1_S=y$. Если она разрешима, базисное решение даёт разложение $y$ не более чем по $T$ спектрам; вместе с леммами~\ref{lem:separation} и~\ref{lem:patch} это явная функция $h$ и мера, реализующие $y$. Если нет, двойственная задача даёт $\lambda\in\R^T$ с $\sum_{t\in S}\lambda_t\ge0$ при всех $S\in\Sp(P)$ и $\ip{\lambda}{y}<0$ (лемма Фаркаша); это опровержение: для любой реализующей пары $(h,\mu)$ из $y=\sum_S\mu(R_S)\mathbf 1_S$ и $\Sp(h,P)\subseteq\Sp(P)$ следовало бы $\ip{\lambda}{y}\ge0$. Неравенство $\sum_{\lambda_t<0}\abs{\lambda_t}y_t\le\sum_{\lambda_t>0}\lambda_ty_t$ выполнено для всех разрешимых $y$ и нарушено для данного.
\end{enumerate}

\begin{example}[проверка неразрешимости]
    Пусть $\hat p_1=(4,1)$, $\hat p_2=(3,3)$, $\hat p_3=(1,4)$ и $y=(1,5,3)$. Шаг~1: единственное покрытие~--- индекс $2$ покрывает пару $1,3$, так как $(2{,}5,\,2{,}5)\le\hat p_2$; поэтому $\Sp(P)=2^{[3]}\setminus\set{\set2}$. Шаг~2: задача $\sum_Sm_S\mathbf 1_S=y$, $m_S\ge0$, неразрешима, и двойственная задача даёт $\lambda=(1,-1,1)$: на каждом $S\in\Sp(P)$ сумма $\sum_{t\in S}\lambda_t$ неотрицательна (множество $\set2$ исключено), а $\ip{\lambda}{y}=-1<0$. Опровержение читается как неравенство $y_2\le y_1+y_3$, верное для всех разрешимых векторов и нарушенное здесь: $5>4$.
\end{example}

\begin{example}
    Опровержения не сводятся к неравенствам $y_t\le y_s+y_r$, отвечающим покрытиям. При $p_1=(107,179)$, $p_2=(59,125)$, $p_3=(26,148)$, $p_4=(66,101)$ покрытия таковы: $p_1$ доминирует $p_2,p_3,p_4$, а $2$ покрывает пару $3,4$. Вектор $y=(1,2,1,1)$ удовлетворяет всем неравенствам покрытий ($y_1\le y_2,y_3,y_4$ и $y_2\le y_3+y_4$), но шаг~2 даёт $\lambda=(-1,-1,1,1)$: всякое $S\in\Sp(P)$, содержащее $1$, содержит $3$ и $4$, а содержащее $2$~--- хотя бы одно из них, так что $\sum_{t\in S}\lambda_t\ge0$, тогда как $\ip{\lambda}{y}=-1$. Это неравенство $y_1+y_2\le y_3+y_4$.
\end{example}

\begin{example}[проверка разрешимости]
    Пусть $\hat p_1=(1,9)$, $\hat p_2=(4,4)$, $\hat p_3=(9,1)$, $\hat p_4=(10,100000)$, $\hat p_5=(20000,20000)$, $\hat p_6=(100000,10)$ и $y=(5,9,6,3,1,4)$. Шаг~1: цены $\hat p_1,\hat p_2,\hat p_3$ попарно несравнимы и не покрывают пар, то же верно для $\hat p_4,\hat p_5,\hat p_6$, а каждая из $\hat p_4,\hat p_5,\hat p_6$ доминирует каждую из $\hat p_1,\hat p_2,\hat p_3$; других покрытий нет. Поэтому $S\in\Sp(P)$ тогда и только тогда, когда из $S\cap\set{4,5,6}\ne\varnothing$ следует $\set{1,2,3}\subset S$: всего $15$ достижимых спектров, восемь подмножеств $\set{1,2,3}$ и семь множеств $\set{1,2,3}\cup U$ с непустым $U\subset\set{4,5,6}$. Шаг~2: задача разрешима,
    \[
        y=4\cdot\mathbf 1_{\set2}+1\cdot\mathbf 1_{\set3}+1\cdot\mathbf 1_{\set{1,2,3,4,5}}+2\cdot\mathbf 1_{\set{1,2,3,6}}+2\cdot\mathbf 1_{\set{1,2,3,4,6}},
    \]
    так что $y$ неоклассически разрешим, и по лемме~\ref{lem:patch} все пять спектров этого разложения реализуются одной кусочно-линейной $h$.
\end{example}

\begin{example}
    Опровержение может не иметь коэффициентов из $\set{-1,0,1}$. Пусть $p_1=(29,799)$, $p_2=(68,763)$, $p_3=(116,484)$, $p_4=(543,388)$, $p_5=(781,787)$, $p_6=(820,172)$; покрытия: $2$ покрывает пару $1,3$; $4$~--- пару $3,6$; $5$~--- пару $1,6$ и доминирует $p_2,p_3,p_4$. Вектор $y=(3,6,3,5,2,3)$ не разрешим, и $\lambda=(1,-1,2,-1,-1,1)$~--- опровержение: $y_2+y_4+y_5\le y_1+2y_3+y_6$ для всех разрешимых $y$, а здесь $13>12$. Перебор всех $3^6$ векторов $\lambda\in\set{-1,0,1}^6$ показывает, что опровержения с такими коэффициентами для этого $y$ не существует: неравенство выше задаёт грань конуса $\cone\set{\mathbf 1_S\mid S\in\Sp(P)}$, и $y$ нарушает только её.
\end{example}

\begin{example}
    Общее положение цен не требуется. При ценах $p_1=(1,3)$, $p_2=(2,2)$, $p_3=(3,1)$, $p_4=(2,3)$ у $p_2$ и $p_4$ совпадают первые координаты, у $p_1$ и $p_4$~--- вторые. Покрытия: $p_4$ доминирует $p_1$ и $p_2$, а $2$ покрывает пару $1,3$. Вектор $y=(2,1,1,1)=\mathbf 1_{\set1}+\mathbf 1_{\set3}+\mathbf 1_{\set{1,2,4}}$ неоклассически разрешим, а $y=(1,1,1,2)$ нет: $\lambda=(1,0,0,-1)$ даёт опровержение $y_4\le y_1$.
\end{example}

\section{Слабая разрешимость}

\begin{lemma}\label{lem:open}
    Пусть $P\in\Rpp[d\times T]$. Тогда существует такая окрестность $V\subset\Rpp[d\times T]$ набора $P$, что $\Sp(Q)\supseteq\Sp(P)$ для каждого набора цен $Q\in V$. В частности, если $y$ разрешим при ценах $P$, то $y$ разрешим при всех ценах из $V$.
\end{lemma}

\begin{proof}
    По теореме~\ref{thm:spectrum} возьмём $h\in\Phi_d$ с $\Sp(h,P)=\Sp(P)$ и окрестность $V$ из леммы~\ref{lem:stability}. При $Q\in V$ по лемме~\ref{lem:necessity} $\Sp(P)=\Sp(h,P)\subseteq\Sp(h,Q)\subseteq\Sp(Q)$. Второе утверждение следует из теоремы~\ref{thm:solvability}.
\end{proof}

\begin{definition}[Слабо достижимые спектры]\label{def:weak}
    Пусть $P:=\set{p_{t}\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Будем говорить, что $S\subset[T]$~--- \textit{слабо достижимый спектр} при ценах $P$, если для каждого $t\in S$ существует ненулевой вектор $\xi\in\Rp[d]$, для которого $\ip{\xi}{p_s-p_t}\ge0$ при всех $s\notin S$. Такой вектор назовём \textit{слабым свидетелем} индекса $t$. Будем обозначать как $\widehat{\Sp}(P)$ семейство таких спектров. Вектор выпусков $y\in\cone\set{\mathbf 1_S\mid S\in\widehat{\Sp}(P)}$ назовём \textit{слабо разрешимым} при $P$.
\end{definition}

\begin{lemma}[слоёный пирог]\label{lem:cake}
    Пусть $z\in\Rp[T]$ и $0=\lambda_0<\lambda_1<\dots<\lambda_k$~--- все различные значения $z$ вместе с нулём. Тогда
    \[
        z=\sum_{j=1}^{k}(\lambda_j-\lambda_{j-1})\,\mathbf 1_{\set{z\ge\lambda_j}}.
    \]
\end{lemma}

\begin{proof}
    При $z_s=\lambda_i$ в правой части остаются слагаемые с $j\le i$, и их сумма телескопируется в $\lambda_i$.
\end{proof}

\begin{lemma}[перепаривание]\label{lem:rematch}
    Пусть $C\subset[T]$~--- набор индексов с одинаковыми ценами, $u\in\widehat{\Sp}(P)$, $u\cap C\ne\varnothing$, $C\setminus u\ne\varnothing$ и $L\subset C$. Тогда $(u\setminus C)\cup L\in\widehat{\Sp}(P)$.
\end{lemma}

\begin{proof}
    Пусть $S':=(u\setminus C)\cup L$. Для $t\in u\setminus C$ подходит его слабый свидетель для $u$: индекс $s\notin S'$ вне $C$ лежит вне $u$, а индекс $s\in C\setminus L$ имеет ту же цену, что некоторый $s_1\in C\setminus u$. Для $t\in L$ подходит слабый свидетель любого $t_1\in u\cap C$ для $u$: цены $p_t$ и $p_{t_1}$ равны, индексы $s\notin S'$ вне $C$ лежат вне $u$, а для $s\in C$ разность $p_s-p_t$ равна нулю.
\end{proof}

\begin{lemma}[монотонное представление]\label{lem:monotone}
    Пусть $y$ слабо разрешим при ценах $P$. Тогда $y=\sum_Sm_S\mathbf 1_S$ при некоторых $m_S\ge0$, $S\in\widehat{\Sp}(P)$, причём для всякого $S$ с $m_S>0$ из $t\in S$, $s\notin S$ и $p_s=p_t$ следует $y_s<y_t$.
\end{lemma}

\begin{proof}
    Кластером назовём множество индексов с одной и той же ценой. Возьмём любое представление $y=\sum_Sm_S\mathbf 1_S$ и исправим его по одному кластеру за раз. Пусть $C$~--- кластер, $U$~--- те $S$ с $m_S>0$, которые пересекают $C$, но не содержат его, $M:=\sum_{u\in U}m_u$ и $z:=\sum_{u\in U}m_u\mathbf 1_{u\cap C}$. Остальные спектры содержат $C$ целиком или не пересекают его, поэтому $y-z$ постоянен на $C$, и верхние множества $z$ на $C$ суть верхние множества $y$ на $C$.
    По лемме~\ref{lem:cake} $z=\sum_{j\ge1}\nu_j\mathbf 1_{L_j}$, где $L_j:=\set{s\in C\mid z_s\ge\lambda_j}$ и $\sum_{j\ge1}\nu_j=\max z\le M$. Положим $L_0:=\varnothing$ и $\nu_0:=M-\max z$, тогда $\sum_{j\ge0}\nu_j=M=\sum_{u\in U}m_u$. Для двух наборов весов с одинаковой суммой $M$ верно тождество $\sum_{u,j}\frac{m_u\nu_j}{M}(a_u+b_j)=\sum_um_ua_u+\sum_j\nu_jb_j$, откуда
    \[
        \sum_{u\in U}m_u\mathbf 1_u=\sum_{u\in U}m_u\mathbf 1_{u\setminus C}+z=\sum_{u\in U}\sum_{j\ge0}\frac{m_u\nu_j}{M}\,\mathbf 1_{(u\setminus C)\cup L_j}.
    \]
    Спектры $(u\setminus C)\cup L_j$ слабо достижимы по лемме~\ref{lem:rematch}, пересекают $C$ по верхнему множеству $y$ и вне $C$ совпадают со старыми. Поэтому свойство, достигнутое для других кластеров, сохраняется.
\end{proof}

\begin{remark}
    Исправлять представление необходимо. При $p_1=p_2=(1,1)$, $p_3=(1,\tfrac12)$, $p_4=(\tfrac12,1)$, $p_5=(\tfrac12,\tfrac32)$, $p_6=(\tfrac32,\tfrac12)$ спектры $\set{1,4,5}$, $\set{1,3,6}$, $\set{2,3,4}$ слабо достижимы, но ни при каких ценах, близких к $P$, не достижимы одновременно.
\end{remark}

\begin{lemma}[наклон]\label{lem:tilt}
    Пусть $P:=\set{p_t\in\Rpp[d]}_{t\in[T]}$~--- набор цен, $y\in\Rp[T]$, и при $\delta>0$ положим
    \[
        Q_\delta:=\set{q_t}_{t\in[T]},\qquad q_t:=p_t-\delta\Bigl(\sum_{i\in[d]}\ln p^i_t+\delta y_t\Bigr)\mathbf e.
    \]
    Пусть $S\in\widehat{\Sp}(P)$ и $y_s<y_t$ при всех $t\in S$, $s\notin S$ с $p_s=p_t$. Тогда $S\in\Sp(Q_\delta)$ при всех достаточно малых $\delta>0$.
\end{lemma}

\begin{proof}
    Пусть $t\in S$, $\xi$~--- слабый свидетель индекса $t$ с $\ip{\xi}{\mathbf e}=1$, $g:=((p^1_t)^{-1},\dots,(p^d_t)^{-1})$, $\xi_\delta:=\xi+\delta g$ и $F_r:=\sum_i\ln p^i_r$. Тогда $q_s-q_t=(p_s-p_t)-\bigl(\delta(F_s-F_t)+\delta^2(y_s-y_t)\bigr)\mathbf e$. Покажем, что $\ip{\xi_\delta}{q_s-q_t}>0$ при $s\notin S$ и малых $\delta$.
    Если $p_s=p_t$, то $\ip{\xi_\delta}{q_s-q_t}=\delta^2(y_t-y_s)\bigl(1+\delta\ip{g}{\mathbf e}\bigr)>0$.
    Если $p_s\ne p_t$, то $\ip{\xi_\delta}{q_s-q_t}=\ip{\xi}{p_s-p_t}+\delta K+O(\delta^2)$ с $K:=\ip{g}{p_s-p_t}-(F_s-F_t)=\sum_i(x_i-1-\ln x_i)$, $x_i:=p^i_s/p^i_t$. Здесь $\ip{\xi}{p_s-p_t}\ge0$, а $K>0$, так как $x-1-\ln x\ge0$ с равенством только при $x=1$.
    Пар $(t,s)$ конечное число, и $\xi_\delta\ge\delta g$ ненулевой, так что $\xi_\delta$~--- свидетель индекса $t$ при ценах $Q_\delta$.
\end{proof}

\begin{theorem}[замыкание]\label{thm:closure}
    Пусть $P:=\set{p_t\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Вектор выпусков $y\in\Rp[T]$ слабо разрешим при ценах $P$ тогда и только тогда, когда в любой открытой окрестности $P$ найдётся набор цен $Q$, при котором $y$ разрешим.
\end{theorem}

\begin{proof}
    Покажем сначала, что $\Sp(Q)\subseteq\widehat{\Sp}(P)$ для всех $Q$, достаточно близких к $P$. Иначе, поскольку спектров конечное число, нашлись бы $S\notin\widehat{\Sp}(P)$, $t\in S$ и наборы $Q_n\to P$ с $S\in\Sp(Q_n)$. Свидетелей $\xi_n$ индекса $t$ при ценах $Q_n$ нормируем условием $\ip{\xi_n}{\mathbf e}=1$. Симплекс компактен, и предел $\xi$ подпоследовательности ненулевой, неотрицательный и удовлетворяет $\ip{\xi}{p_s-p_t}\ge0$ при $s\notin S$, то есть является слабым свидетелем. Это противоречит выбору $S$ и $t$.
    Пусть теперь в окрестности $P$, где $\Sp(Q)\subseteq\widehat{\Sp}(P)$, есть $Q$, при котором $y$ разрешим. По теореме~\ref{thm:solvability} $y\in\cone\set{\mathbf 1_S\mid S\in\Sp(Q)}\subseteq\cone\set{\mathbf 1_S\mid S\in\widehat{\Sp}(P)}$.
    Обратно, пусть $y$ слабо разрешим. Возьмём представление из леммы~\ref{lem:monotone}. По лемме~\ref{lem:tilt} все его спектры, которых конечное число, лежат в $\Sp(Q_\delta)$ при малых $\delta>0$. Тогда $y$ разрешим при ценах $Q_\delta$ по теореме~\ref{thm:solvability}, а $Q_\delta\to P$ при $\delta\to0$, так что $Q_\delta$ попадает в любую окрестность $P$.
\end{proof}

\section{Минимальное возмущение цен}

\begin{definition}[Возмущение цен]\label{def:perturbation}
    Для $p,q>0$ назовём \textit{возмущением} величину
    \[
        \pi(p,q):=\max\set{\abs{q/p-1},\ \abs{p/q-1}}.
    \]
    Для наборов цен $P:=\set{p_t\in\Rpp[d]}_{t\in[T]}$ и $Q:=\set{q_t\in\Rpp[d]}_{t\in[T]}$ положим $\pi(P,Q):=\sum_{t\in[T],\,i\in[d]}\pi(p^i_t,q^i_t)$. Для вектора выпусков $y\in\Rp[T]$ назовём \textit{минимальным возмущением} набора цен $P$ величину
    \[
        \rho(P,y):=\inf\set{\pi(P,Q)\mid Q\in\Rpp[d\times T],\ y\ \text{разрешим при ценах}\ Q}.
    \]
    Это наименьшее относительное изменение цен, при котором $y$ становится разрешим. Возмущение симметрично, $\pi(p,q)=\pi(q,p)$, и $\pi(p,q)\to\infty$ при $q\to0$.
\end{definition}

\begin{remark}
    По лемме~\ref{lem:open} множество наборов цен, при которых $y$ разрешим, открыто. Поэтому при $\rho(P,y)>0$ точная нижняя грань не достигается: у допустимого набора $Q$ есть целая окрестность допустимых, и в ней найдётся $Q'$ с меньшим возмущением. Ниже задача дополняется запасом $W$, который делает минимум достижимым.
\end{remark}

\begin{lemma}[Каратеодори]\label{lem:cara}
    Пусть $\mathcal A\subset2^{[T]}$ и $y\in\cone\set{\mathbf 1_S\mid S\in\mathcal A}$. Тогда $y=\sum_{k\in[T]}m_k\mathbf 1_{S_k}$ при некоторых $m_k\ge0$ и $S_k\in\mathcal A$.
\end{lemma}

\begin{proof}
    Среди представлений $y=\sum_{S\in\mathcal A}m_S\mathbf 1_S$ с $m_S\ge0$ возьмём представление с наименьшим числом ненулевых коэффициентов. Если их больше $T$, векторы $\mathbf 1_S$ с $m_S>0$ линейно зависимы в $\R^T$: $\sum_Sg_S\mathbf 1_S=0$, $g\ne0$, и, меняя знак, считаем, что некоторое $g_S>0$. Положим $\theta:=\min_{g_S>0}m_S/g_S$. Коэффициенты $m_S-\theta g_S$ неотрицательны, дают то же $y$ и обращаются в нуль ещё в одном месте. Противоречие с минимальностью.
\end{proof}

\begin{definition}[Задача о возмущении]\label{def:model}
    Пусть $P:=\set{p_t\in\Rpp[d]}_{t\in[T]}$~--- набор цен и $y\in\Rp[T]$~--- вектор выпусков. Зафиксируем параметры $D>0$ и $W>0$: границу возмущения цен и границу свидетелей. По ним рассчитаем вспомогательные границы
    \[
        Y:=\max_{t\in[T]}y_t,\qquad M_{st}:=1+W(1+D)\sum_{i\in[d]}(p^i_s+p^i_t).
    \]
    Переменные:
    \begin{itemize}
        \item $a^i_t$~--- множители цен, $q^i_t:=p^i_ta^i_t$;
        \item $b^i_t$ и $g^i_t$~--- обратные множители и возмущения координат;
        \item $z_{k,t}$~--- индикаторы спектров $S_k:=\set{t\mid z_{k,t}=1}$;
        \item $m_k$ и $w_{k,t}$~--- массы и их произведения с индикаторами;
        \item $\xi^i_{k,t}$~--- свидетели.
    \end{itemize}
    Границы на переменные~--- техническое требование решателя: в ограниченной коробке пространственный метод ветвей и границ сходится и сертифицирует глобальный оптимум. \textit{Задачей о возмущении} с параметрами $(D,W)$ назовём задачу
    \begin{empheq}[left=\empheqlbrace]{align}
        & \sum_{t\in[T],\,i\in[d]}g^i_t\ \to\ \min, \label{eq:obj}\\
        & g^i_t\ge a^i_t-1,\quad g^i_t\ge b^i_t-1,\quad a^i_tb^i_t\ge1, & &\forall\,t\in[T],\ i\in[d], \label{eq:pert}\\
        & \sum_{k\in[T]}w_{k,t}=y_t, & &\forall\,t\in[T], \label{eq:cone}\\
        & w_{k,t}\le Y z_{k,t},\quad w_{k,t}\le m_k,\quad w_{k,t}\ge m_k-Y(1-z_{k,t}), & &\forall\,k,t\in[T], \label{eq:product}\\
        & \sum_{i\in[d]}\xi^i_{k,t}\,\bigl(p^i_sa^i_s-p^i_ta^i_t\bigr)\ \ge\ 1-M_{st}\,(1-z_{k,t}+z_{k,s}), & &\forall\,k,s,t\in[T],\ s\ne t, \label{eq:witness}\\
        & a^i_t,\,b^i_t\in[0,\,1+D],\quad g^i_t\in[0,D],\quad \xi^i_{k,t}\in[0,W], & &\forall\,k,t\in[T],\ i\in[d], \label{eq:box}\\
        & z_{k,t}\in\set{0,1},\quad m_k\in[0,Y],\quad w_{k,t}\in[0,Y], & &\forall\,k,t\in[T]. \label{eq:box2}
    \end{empheq}
    Её оптимальное значение обозначим $v(D,W)$.
\end{definition}

\begin{lemma}[полный спектр]\label{lem:full}
    Пусть $P:=\set{p_t\in\Rpp[d]}_{t\in[T]}$~--- набор цен, у которого первые координаты $p^1_t$ попарно различны, а произведение первых двух координат одно и то же: $p^1_tp^2_t=\kappa$ при всех $t\in[T]$. Тогда всякое $S\subset[T]$ является достижимым спектром, то есть $\Sp(P)=2^{[T]}$. В частности, всякий вектор выпусков $y\in\Rp[T]$ разрешим при ценах $P$.
\end{lemma}

\begin{proof}
    Положим $\xi_t:=\bigl((p^1_t)^{-1},(p^2_t)^{-1},0,\dots,0\bigr)$. Так как $p^2_s/p^2_t=p^1_t/p^1_s$, при $s\ne t$
    \[
        \ip{\xi_t}{p_s-p_t}=\frac{p^1_s}{p^1_t}+\frac{p^1_t}{p^1_s}-2=\frac{(p^1_s-p^1_t)^2}{p^1_sp^1_t}>0.
    \]
    Поэтому $\xi_t$ служит свидетелем индекса $t$ для всякого $S\ni t$, и $S\in\Sp(P)$. Второе утверждение: $y=\sum_ty_t\mathbf 1_{\set t}\in\cone\set{\mathbf 1_S\mid S\in\Sp(P)}$, и $y$ разрешим по теореме~\ref{thm:solvability}.
\end{proof}

\begin{lemma}[коробка]\label{lem:box}
    Пусть $p,q>0$ и $D\ge0$. Тогда $\pi(p,q)=\max\set{q/p,\,p/q}-1$, и $\pi(p,q)\le D$ тогда и только тогда, когда $p/(1+D)\le q\le p(1+D)$. В частности, если $\pi(P,Q)\le D$, то $p^i_t/(1+D)\le q^i_t\le p^i_t(1+D)$ при всех $t\in[T]$, $i\in[d]$.
\end{lemma}

\begin{proof}
    Положим $a:=q/p$. При $a\ge1$ имеем $\abs{a-1}=a-1\ge1-a^{-1}=\abs{a^{-1}-1}$, при $a\le1$~--- симметрично. Поэтому $\pi(p,q)=\max\set{a,a^{-1}}-1$, и $\pi(p,q)\le D$ равносильно $a\le1+D$ и $a^{-1}\le1+D$. Второе утверждение: слагаемые в $\pi(P,Q)$ неотрицательны, поэтому $\pi(p^i_t,q^i_t)\le\pi(P,Q)\le D$.
\end{proof}

\begin{lemma}[оценка сверху]\label{lem:bound}
    Пусть $d\ge2$, $P:=\set{p_t\in\Rpp[d]}_{t\in[T]}$~--- набор цен, $y\in\Rp[T]$, $\kappa>0$ и
    \[
        D(\kappa):=\sum_{t\in[T]}\Bigl(\max\set{\frac{\kappa}{p^1_tp^2_t},\ \frac{p^1_tp^2_t}{\kappa}}-1\Bigr).
    \]
    Тогда для всякого $\eta>0$ найдётся набор цен $Q\in\Rpp[d\times T]$, при котором $y$ разрешим, с $\pi(P,Q)\le D(\kappa)+\eta$. В частности, $\rho(P,y)\le D(\kappa)$.
\end{lemma}

\begin{proof}
    Для $\theta\ge0$ положим $q^1_t:=p^1_t+\theta t$, $q^2_t:=\kappa/q^1_t$ и $q^i_t:=p^i_t$ при $i>2$. Цены $Q_\theta$ положительны, $q^1_tq^2_t=\kappa$, а $q^1_t$ попарно различны, если $\theta\ne(p^1_t-p^1_s)/(s-t)$ при всех $s\ne t$. Функция $\theta\mapsto\pi(P,Q_\theta)$ непрерывна, а при $\theta=0$ по лемме~\ref{lem:box} равна
    \[
        \sum_{t\in[T]}\pi\bigl(p^2_t,\kappa/p^1_t\bigr)=D(\kappa).
    \]
    Поэтому при малом $\theta>0$ вне конечного множества исключений $\pi(P,Q_\theta)\le D(\kappa)+\eta$, а $y$ разрешим при ценах $Q_\theta$ по лемме~\ref{lem:full}.
\end{proof}

\begin{proposition}\label{prop:bounds}
    Пусть $d\ge2$ и $D>D(\kappa)$ при некотором $\kappa>0$. Тогда $\rho(P,y)<D$, и для всякого набора цен $Q$ с $\pi(P,Q)\le D$ значения $a^i_t:=q^i_t/p^i_t$, $b^i_t:=p^i_t/q^i_t$, $g^i_t:=\pi(p^i_t,q^i_t)$ удовлетворяют~\eqref{eq:pert} и~\eqref{eq:box}. Таким образом, границы на $a$, $b$ и $g$ не исключают ни одного набора цен с возмущением не больше $D$.
\end{proposition}

\begin{proof}
    Первое утверждение~--- лемма~\ref{lem:bound}. Далее, $a^i_tb^i_t=1$, и по лемме~\ref{lem:box} $a^i_t,b^i_t\in[1/(1+D),1+D]\subset[0,1+D]$ и $g^i_t=\max\set{a^i_t,b^i_t}-1\in[0,D]$.
\end{proof}

\begin{proposition}[внутренняя точка]\label{prop:interior}
    Пусть $d\ge2$, $Q:=\set{q_t\in\Rpp[d]}_{t\in[T]}$~--- набор цен с попарно различными $q^1_t$ и $q^1_tq^2_t=\kappa$ (например, $Q=Q_\theta$ из леммы~\ref{lem:bound}), и
    \[
        W_{\mathrm{crit}}(Q):=\max_{t\in[T]}c_t\max\set{(q^1_t)^{-1},(q^2_t)^{-1}},\qquad c_t:=\max_{s\ne t}\frac{q^1_sq^1_t}{(q^1_s-q^1_t)^2}.
    \]
    Если $D\ge\pi(P,Q)$ и $W\ge W_{\mathrm{crit}}(Q)$, то задача о возмущении с параметрами $(D,W)$ имеет допустимую точку со значением целевой функции $\pi(P,Q)$. В частности, $v(D,W)\le\pi(P,Q)$.
\end{proposition}

\begin{proof}
    Положим $a^i_t:=q^i_t/p^i_t$, $b^i_t:=p^i_t/q^i_t$, $g^i_t:=\pi(p^i_t,q^i_t)$, $z_{k,t}:=[k=t]$, $m_k:=y_k$, $w_{k,t}:=y_k[k=t]$ и $\xi_{k,t}:=\xi_t:=c_t\bigl((q^1_t)^{-1},(q^2_t)^{-1},0,\dots,0\bigr)$. Ограничения~\eqref{eq:pert} и~\eqref{eq:box} для $a$, $b$, $g$ выполнены по лемме~\ref{lem:box}, ограничения~\eqref{eq:cone}, \eqref{eq:product}, \eqref{eq:box2}~--- непосредственно. Далее, $0\le\xi_t\le c_t\max\set{(q^1_t)^{-1},(q^2_t)^{-1}}\le W$. В~\eqref{eq:witness} при $k=t$ и $s\ne t$ правая часть равна $1$, а левая по вычислению из леммы~\ref{lem:full} равна $c_t(q^1_s-q^1_t)^2/(q^1_sq^1_t)\ge1$. В остальных случаях $1-z_{k,t}+z_{k,s}\ge1$, правая часть не больше $1-M_{st}$, а левая не меньше $-W\sum_iq^i_t\ge-W(1+D)\sum_ip^i_t\ge1-M_{st}$.
\end{proof}

\begin{remark}
    Величины $D(\kappa)$ и $W_{\mathrm{crit}}(Q_\theta)$ вычисляются до запуска решателя. Если первые координаты $p^1_t$ попарно различны, можно взять $\theta=0$: тогда $\pi(P,Q_0)=D(\kappa)$. При $\theta\to0$ величина $W_{\mathrm{crit}}(Q_\theta)$ растёт, если у $P$ есть совпадающие первые координаты.
\end{remark}

\begin{remark}
    Задача~\eqref{eq:obj}--\eqref{eq:box2}~--- смешанно-целочисленная задача с линейной целевой функцией, бинарными переменными $z$ и билинейными членами $\xi^i_{k,t}a^i_s$ в~\eqref{eq:witness}. Множество $\set{ab\ge1,\ a,b\ge0}$ выпукло (повёрнутый конус второго порядка), так что~\eqref{eq:pert} невыпуклости не добавляет. Из $a^i_tb^i_t\ge1$ и $b^i_t\le1+D$ следует $a^i_t\ge1/(1+D)$, так что в~\eqref{eq:box} нижняя граница нуль ничего не теряет; в коде границу $a^i_t,b^i_t\ge1/(1+D)$ стоит задать явно, оболочки Маккормика для $\xi^i_{k,t}a^i_s$ от этого становятся точнее. Все переменные ограничены. Такие задачи Gurobi решает до глобального оптимума пространственным методом ветвей и границ с оболочками Маккормика для билинейных членов~\cite{Gurobi}. Величина \texttt{MIPGap} относится к глобальной нижней оценке, поэтому статус \texttt{OPTIMAL} сертифицирует глобальный оптимум $v(D,W)$ с точностью до допусков решателя, а статус \texttt{INFEASIBLE}~--- отсутствие решений в коробке с данным запасом $W$. Билинейных членов $dT^3$, бинарных переменных $T^2$. Спектры $S_k$ взаимозаменяемы, поэтому полезно добавить $m_1\ge\dots\ge m_T$.
\end{remark}

\begin{remark}
    Допуски решателя снимаются точной проверкой найденного решения. Веса $\xi_{k,t}$ округляются до рациональных чисел; запас в единицу в~\eqref{eq:witness} позволяет проверить неравенства $\ip{\xi_{k,t}}{q_s-q_t}\ge1$ в точной арифметике. Массы $m_k$ пересчитываются точно решением системы $\sum_km_k\mathbf 1_{S_k}=y$ по найденным $S_k$.
\end{remark}
