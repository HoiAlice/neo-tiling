\section{Разрешимость}

Всюду $d\in\N$, $d\ge2$. Для $p,x\in\R^d$ через $p\circ x:=(p^1x^1,\dots,p^dx^d)$ обозначим покоординатное произведение, через $\mathbf e:=(1,\dots,1)\in\R^d$~--- единичный вектор, через $\mathbf 1_S\in\R^T$~--- индикатор множества $S\subset[T]$. Неравенства между векторами покоординатные.

\begin{definition}[Неоклассические Функции]\label{def:neoclassical}
    Будем называть функцию $f:\Rp[d]\to\R_{+}$ \textit{неоклассической}, если она \textit{непрерывна}, \textit{вогнута} и однородна степени~1.\\
    Будем обозначать класс неоклассических функций $d$ переменных как $\Phi_d$.
\end{definition}

\begin{definition}[Неоклассическая разрешимость]\label{def:neo-solvability}
    Пусть $\set{\hat p_t\in\Rpp[d]}_{t\in[T]}$ --- некоторый конечный набор цен.
    Будем говорить, что вектор выпусков $y\in\Rp[T]$ \textit{неоклассически разрешим} при этих ценах, если
    существует такая неоклассическая функция $h\in\Phi_{d}$ и такие окрестности
    $\set{U_t\ni\hat p_t}_{t\in[T]}$, $U_t\subset\Rpp[d]$, что для каждого набора $\set{p_t\in U_t}_{t\in[T]}$
    существует абсолютно непрерывная мера $\mu$ на $\Rp[d]$, что $\forall t\in[T]\Rightarrow\mu(A_t)=y_t$, где ${A_t=\set{x\in\Rpp[d]\mid h(p_t\circ x)<1}}.$
\end{definition}

\section{Спектр}

\begin{definition}[Спектр]\label{def:spectrum}
    Пусть $h\in\Phi_d$ и $P:=\set{p_t\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Для $S\subset[T]$ назовём \textit{камерой} множество
    \[
        R_S:=\set{x\in\Rpp[d]\mid h(p_t\circ x)<1\ \text{при } t\in S,\ h(p_s\circ x)>1\ \text{при } s\notin S}.
    \]
    \textit{Спектром} пары $(h,P)$ назовём семейство $\Sp(h,P):=\set{S\subset[T]\mid R_S\neq\varnothing}$.
    Камеры открыты по непрерывности $h$ и попарно не пересекаются. Точка $x\in\Rpp[d]$ вне кривых $h(p_t\circ x)=1$ лежит в камере $R_{S(x)}$, $S(x):=\set{t\mid h(p_t\circ x)<1}$, поэтому $A_t:=\set{x\in\Rpp[d]\mid h(p_t\circ x)<1}$ отличается от $\bigsqcup_{S\ni t}R_S$ лишь точками кривых.
\end{definition}

\begin{lemma}\label{lem:null}
    Пусть $h\in\Phi_d$ и $p\in\Rpp[d]$. Тогда кривая $\Gamma:=\set{x\in\Rpp[d]\mid h(p\circ x)=1}$ имеет меру нуль.
\end{lemma}

\begin{proof}
    По однородности при $c>0$ множество $\set{x\in\Rpp[d]\mid h(p\circ x)=c}$ равно $c\Gamma$ и имеет меру $c^d\abs\Gamma$. Эти множества при разных $c$ попарно не пересекаются, поэтому положительную меру могут иметь лишь счётное их число. Значит, $\abs\Gamma=0$.
\end{proof}

\begin{lemma}\label{lem:signature}
    Рассмотрим неоклассическую функцию $h\in\Phi_{d}$ и набор цен $P:=\set{p_t\in\Rpp[d]}_{t\in[T]}$. Тогда вектор выпусков $y\in\Rp[T]$ принадлежит конусу $\cone\set{\mathbf 1_S\mid S\in\Sp(h,P)}$, если и только если существует абсолютно непрерывная мера $\mu$ на $\Rp[d]$, для которой при всех $t\in[T]$ выполнено интегральное условие $\mu(A_t)=y_t$, где $A_t:=\set{x\in\Rpp[d]\mid h(p_t\circ x)<1}$.
\end{lemma}

\begin{proof}
    По лемме~\ref{lem:null} кривые имеют меру нуль, поэтому $\mu(A_t)=\sum_{S\ni t}\mu(R_S)$, и условия $\mu(A_t)=y_t$ означают $y=\sum_S\mu(R_S)\mathbf 1_S$, где $\mu(R_S)=0$ при $S\notin\Sp(h,P)$. Обратно, при $S\in\Sp(h,P)$ камера $R_S$~--- непустое открытое множество, и равномерная вероятностная мера на шаре в $R_S$ реализует вектор $\mathbf 1_S$, а суммы и неотрицательные кратные мер реализуют суммы и кратные векторов.
\end{proof}

\begin{lemma}\label{lem:stability}
    Пусть $h\in\Phi_d$, $P\in\Rpp[d\times T]$. Тогда существует такая окрестность $V\subset\Rpp[d\times T]$ набора $P$, что для каждого набора цен $Q\in V$ выполнено $\Sp(h,Q)\supseteq\Sp(h,P)$.
\end{lemma}

\begin{proof}
    Пусть $S\in\Sp(h,P)$ и $x\in R_S$. Строгие неравенства $h(p_t\circ x)\gtrless1$ сохраняются при малом изменении цен по непрерывности $h$, поэтому $x$ лежит в камере со спектром $S$ и для $Q$, то есть $S\in\Sp(h,Q)$. Множеств $S$ конечное число.
\end{proof}

\section{Достижимые спектры}

\begin{definition}[Достижимые спектры]\label{def:reachable}
    Пусть $P:=\set{p_{t}\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Будем говорить, что $S\subset[T]$~--- \textit{достижимый спектр} при ценах $P$, если для каждого $t\in S$ существует ненулевой вектор $\xi\in\Rp[d]$, для которого $\ip{\xi}{p_s-p_t}>0$ при всех $s\notin S$. Такой вектор назовём \textit{свидетелем} индекса $t$. Будем обозначать как $\Sp(P)$ семейство таких спектров.
\end{definition}

\begin{lemma}\label{lem:necessity}
    Пусть $h\in\Phi_{d}$ и $P:=\set{p_{t}\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Тогда $\Sp(h,P)\subseteq\Sp(P)$.
\end{lemma}

\begin{proof}
    Пусть $S\in\Sp(h,P)$, $x\in R_S$ и $t\in S$. Если $S=[T]$, свидетелем служит любой ненулевой $\xi\in\Rp[d]$. Иначе множество $C:=\conv\set{p_s\circ x}_{s\notin S}+\Rp[d]$ выпукло и замкнуто как сумма компакта и замкнутого конуса. На $C$ выполнено $h\ge1$: при $r\notin S$ имеем $h(p_r\circ x)\ge1$, и по монотонности $h$ и неравенству Йенсена
    \[
        h\Bigl(\sum_{r\notin S}w_r\,p_r\circ x+v\Bigr)\ge h\Bigl(\sum_{r\notin S}w_r\,p_r\circ x\Bigr)\ge\sum_{r\notin S}w_r\,h(p_r\circ x)\ge1
    \]
    при $w_r\ge0$, $\sum_rw_r=1$, $v\ge0$. Так как $h(p_t\circ x)<1$, точка $p_t\circ x$ не лежит в $C$. По теореме об отделимости найдутся линейная функция $f$ и число $u$, что $f(p_t\circ x)<u<f(c)$ при всех $c\in C$. Функция $f$ неотрицательна на $\Rp[d]$: иначе при $f(v)<0$, $v\ge0$, точка $c+\lambda v\in C$ с подходящим $\lambda\ge0$ дала бы $f(c+\lambda v)=u$. Положим $\xi^i:=f(e_i)x^i\ge0$. Тогда $\ip{\xi}{p}=f(p\circ x)$, и при $s\notin S$ 
    $\ip{\xi}{p_s-p_t}=f(p_s\circ x)-f(p_t\circ x)>0$,
    откуда также $\xi\ne0$.
\end{proof}

\section{Критерий разрешимости}

\begin{lemma}[отделимость]\label{lem:separation}
    Пусть $S\in\Sp(P)$ и $t\in S$. Тогда существует вектор $\xi\in\Rpp[d]$, для которого $\ip{\xi}{p_t}<1<\ip{\xi}{p_s}$ при всех $s\notin S$.
\end{lemma}

\begin{proof}
    Пусть $\xi_0$~--- свидетель индекса $t$. При достаточно малом $\delta>0$ вектор $\xi_1:=\xi_0+\delta\mathbf e\in\Rpp[d]$ всё ещё удовлетворяет $\ip{\xi_1}{p_s-p_t}>0$ при всех $s\notin S$, так как таких $s$ конечное число. Положим $a:=\ip{\xi_1}{p_t}>0$ и выберем $c$ с $a<c<\min_{s\notin S}\ip{\xi_1}{p_s}$, а при $S=[T]$ положим $c:=a+1$. Тогда $\xi:=\xi_1/c$ искомый.
\end{proof}

\begin{lemma}[склейка]\label{lem:patch}
    Пусть $P:=\set{p_{t}\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Тогда существует $h\in\Phi_d$, для которой $\Sp(P)\subseteq\Sp(h,P)$.
\end{lemma}

\begin{proof}
    Для каждой пары $(S,t)$ с $t\in S\in\Sp(P)$ возьмём по лемме~\ref{lem:separation} вектор $\eta_{S,t}\in\Rpp[d]$ с $\ip{\eta_{S,t}}{p_t}<1$ и $\ip{\eta_{S,t}}{p_s}>1$ при $s\notin S$, и добавим к ним вектор $\eta_0:=\mathbf e$. Каждому $S\in\Sp(P)$ дадим свой показатель $E_S\in\N$, $E_S\ge1$, а вектору $\eta_0$~--- показатель $0$. Выберем $M\ge1$ так, чтобы $M\min(\eta^1p^1_s,\ \eta^2p^2_s)>1$ для всех выбранных векторов $\eta$ и всех $s\in[T]$, и для $z\in\Z$ положим $x_z:=(M^z,M^{-z},1,\dots,1)$. Тогда при $z\ne0$
    \[
        \ip{\eta}{p\circ x_z}\ \ge\ \eta^1p^1M^z+\eta^2p^2M^{-z}\ \ge\ M\min(\eta^1p^1,\ \eta^2p^2)>1,
    \]
    поскольку одно из чисел $M^z$, $M^{-z}$ не меньше $M$, а оба слагаемых положительны. Положим $\tilde\eta_{S,t}:=\eta_{S,t}\circ x_{-E_S}$, $\tilde\eta_0:=\eta_0$ и
    \[
        h(x):=\min\bigl(\ip{\tilde\eta_0}{x},\ \min_{t\in S\in\Sp(P)}\ip{\tilde\eta_{S,t}}{x}\bigr)\in\Phi_d.
    \]
    Так как $\ip{\eta\circ x_{-a}}{p\circ x_b}=\ip{\eta}{p\circ x_{b-a}}$, в точке $x_{E_S}$ векторы с показателем $E_S$ дают значения $\ip{\eta_{S,t}}{p}$, а все остальные~--- значения $\ip{\eta}{p\circ x_z}$ с $z\ne0$, большие единицы. Поэтому для $t\in S$ имеем $h(p_t\circ x_{E_S})\le\ip{\eta_{S,t}}{p_t}<1$, а для $s\notin S$~--- $h(p_s\circ x_{E_S})>1$. Значит, $x_{E_S}\in R_S$ и $S\in\Sp(h,P)$.
\end{proof}

\begin{theorem}\label{thm:spectrum}
    Пусть $P:=\set{p_{t}\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Тогда существует $h\in\Phi_d$, для которой $\Sp(h,P)=\Sp(P)$.
\end{theorem}

\begin{proof}
    Включение $\Sp(P)\subseteq\Sp(h,P)$ даёт лемма~\ref{lem:patch}, обратное~--- лемма~\ref{lem:necessity}.
\end{proof}

\begin{theorem}\label{thm:solvability}
    Пусть $P:=\set{p_{t}\in\Rpp[d]}_{t\in[T]}$~--- набор цен. Тогда вектор выпусков $y\in\Rp[T]$ неоклассически разрешим при ценах $P$ тогда и только тогда, когда $y\in\cone\set{\mathbf 1_S\mid S\in\Sp(P)}$.
\end{theorem}

\begin{proof}
    Пусть $y$ разрешим с функцией $h\in\Phi_d$. Взяв в определении~\ref{def:neo-solvability} сами цены $P$, по лемме~\ref{lem:signature} получаем $y\in\cone\set{\mathbf 1_S\mid S\in\Sp(h,P)}$, а $\Sp(h,P)\subseteq\Sp(P)$ по лемме~\ref{lem:necessity}. Обратно, возьмём $h$ из теоремы~\ref{thm:spectrum}. По лемме~\ref{lem:stability} $\Sp(h,Q)\supseteq\Sp(h,P)=\Sp(P)$ для всех $Q$ из некоторой окрестности $P$, и по лемме~\ref{lem:signature} при каждом таком $Q$ существует абсолютно непрерывная мера $\mu$ с $\mu(A_t)=y_t$.
\end{proof}

\begin{remark}
    Функция $h$ из теоремы~\ref{thm:spectrum} не зависит от $y$. Тем самым каждый разрешимый вектор выпусков реализуется одной и той же функцией $h$, и от $y$ зависит только мера. Никаких условий общего положения на набор цен не требуется: совпадающие координаты и даже совпадающие цены допустимы.
\end{remark}

\paragraph{Алгоритм.} Теорема~\ref{thm:solvability} даёт конечную процедуру проверки неоклассической разрешимости $y$ при ценах $P$.
\begin{enumerate}
    \item \emph{Построение $\Sp(P)$.} Перебираем $S\subset[T]$ и для каждого $t\in S$ ищем свидетеля, то есть решаем задачу линейного программирования из замечания после определения~\ref{def:reachable}. Оставляем те $S$, у которых свидетели есть у всех $t\in S$. Это $2^T$ кандидатов и не более $T$ задач линейного программирования на каждый.
    \item \emph{Проверка принадлежности конусу.} Решаем задачу линейного программирования: найти $m_S\ge0$, $S\in\Sp(P)$, с $\sum_Sm_S\mathbf 1_S=y$. Если она разрешима, базисное решение даёт разложение $y$ не более чем по $T$ спектрам; вместе с леммами~\ref{lem:separation} и~\ref{lem:patch} это явная функция $h$ и мера, реализующие $y$. Если нет, двойственная задача даёт $\lambda\in\R^T$ с $\sum_{t\in S}\lambda_t\ge0$ при всех $S\in\Sp(P)$ и $\ip{\lambda}{y}<0$ (лемма Фаркаша); это опровержение: для любой реализующей пары $(h,\mu)$ из $y=\sum_S\mu(R_S)\mathbf 1_S$ и $\Sp(h,P)\subseteq\Sp(P)$ следовало бы $\ip{\lambda}{y}\ge0$. Неравенство $\sum_{\lambda_t<0}\abs{\lambda_t}y_t\le\sum_{\lambda_t>0}\lambda_ty_t$ выполнено для всех разрешимых $y$ и нарушено для данного.
\end{enumerate}

\begin{example}[проверка неразрешимости]
    Пусть $\hat p_1=(4,1)$, $\hat p_2=(3,3)$, $\hat p_3=(1,4)$ и $y=(1,5,3)$. Шаг~1: единственное покрытие~--- индекс $2$ покрывает пару $1,3$, так как $(2{,}5,\,2{,}5)\le\hat p_2$; поэтому $\Sp(P)=2^{[3]}\setminus\set{\set2}$. Шаг~2: задача $\sum_Sm_S\mathbf 1_S=y$, $m_S\ge0$, неразрешима, и двойственная задача даёт $\lambda=(1,-1,1)$: на каждом $S\in\Sp(P)$ сумма $\sum_{t\in S}\lambda_t$ неотрицательна (множество $\set2$ исключено), а $\ip{\lambda}{y}=-1<0$. Опровержение читается как неравенство $y_2\le y_1+y_3$, верное для всех разрешимых векторов и нарушенное здесь: $5>4$.
\end{example}

\begin{example}
    Опровержения не сводятся к неравенствам $y_t\le y_s+y_r$, отвечающим покрытиям. При $p_1=(107,179)$, $p_2=(59,125)$, $p_3=(26,148)$, $p_4=(66,101)$ покрытия таковы: $p_1$ доминирует $p_2,p_3,p_4$, а $2$ покрывает пару $3,4$. Вектор $y=(1,2,1,1)$ удовлетворяет всем неравенствам покрытий ($y_1\le y_2,y_3,y_4$ и $y_2\le y_3+y_4$), но шаг~2 даёт $\lambda=(-1,-1,1,1)$: всякое $S\in\Sp(P)$, содержащее $1$, содержит $3$ и $4$, а содержащее $2$~--- хотя бы одно из них, так что $\sum_{t\in S}\lambda_t\ge0$, тогда как $\ip{\lambda}{y}=-1$. Это неравенство $y_1+y_2\le y_3+y_4$.
\end{example}

\begin{example}[проверка разрешимости]
    Пусть $\hat p_1=(1,9)$, $\hat p_2=(4,4)$, $\hat p_3=(9,1)$, $\hat p_4=(10,100000)$, $\hat p_5=(20000,20000)$, $\hat p_6=(100000,10)$ и $y=(5,9,6,3,1,4)$. Шаг~1: цены $\hat p_1,\hat p_2,\hat p_3$ попарно несравнимы и не покрывают пар, то же верно для $\hat p_4,\hat p_5,\hat p_6$, а каждая из $\hat p_4,\hat p_5,\hat p_6$ доминирует каждую из $\hat p_1,\hat p_2,\hat p_3$; других покрытий нет. Поэтому $S\in\Sp(P)$ тогда и только тогда, когда из $S\cap\set{4,5,6}\ne\varnothing$ следует $\set{1,2,3}\subset S$: всего $15$ достижимых спектров, восемь подмножеств $\set{1,2,3}$ и семь множеств $\set{1,2,3}\cup U$ с непустым $U\subset\set{4,5,6}$. Шаг~2: задача разрешима,
    \[
        y=4\cdot\mathbf 1_{\set2}+1\cdot\mathbf 1_{\set3}+1\cdot\mathbf 1_{\set{1,2,3,4,5}}+2\cdot\mathbf 1_{\set{1,2,3,6}}+2\cdot\mathbf 1_{\set{1,2,3,4,6}},
    \]
    так что $y$ неоклассически разрешим, и по лемме~\ref{lem:patch} все пять спектров этого разложения реализуются одной функцией $h\in\Phi_d$.
\end{example}

\begin{example}
    Опровержение может не иметь коэффициентов из $\set{-1,0,1}$. Пусть $p_1=(29,799)$, $p_2=(68,763)$, $p_3=(116,484)$, $p_4=(543,388)$, $p_5=(781,787)$, $p_6=(820,172)$; покрытия: $2$ покрывает пару $1,3$; $4$~--- пару $3,6$; $5$~--- пару $1,6$ и доминирует $p_2,p_3,p_4$. Вектор $y=(3,6,3,5,2,3)$ не разрешим, и $\lambda=(1,-1,2,-1,-1,1)$~--- опровержение: $y_2+y_4+y_5\le y_1+2y_3+y_6$ для всех разрешимых $y$, а здесь $13>12$. Перебор всех $3^6$ векторов $\lambda\in\set{-1,0,1}^6$ показывает, что опровержения с такими коэффициентами для этого $y$ не существует: неравенство выше задаёт грань конуса $\cone\set{\mathbf 1_S\mid S\in\Sp(P)}$, и $y$ нарушает только её.
\end{example}

\begin{example}
    Общее положение цен не требуется. При ценах $p_1=(1,3)$, $p_2=(2,2)$, $p_3=(3,1)$, $p_4=(2,3)$ у $p_2$ и $p_4$ совпадают первые координаты, у $p_1$ и $p_4$~--- вторые. Покрытия: $p_4$ доминирует $p_1$ и $p_2$, а $2$ покрывает пару $1,3$. Вектор $y=(2,1,1,1)=\mathbf 1_{\set1}+\mathbf 1_{\set3}+\mathbf 1_{\set{1,2,4}}$ неоклассически разрешим, а $y=(1,1,1,2)$ нет: $\lambda=(1,0,0,-1)$ даёт опровержение $y_4\le y_1$.
\end{example}

\section{Слабая разрешимость}

\begin{lemma}\label{lem:open}
    Пусть $P\in\Rpp[d\times T]$. Тогда существует такая окрестность $V\subset\Rpp[d\times T]$ набора $P$, что $\Sp(Q)\supseteq\Sp(P)$ для каждого набора цен $Q\in V$.
\end{lemma}

\begin{proof}
    По теореме~\ref{thm:spectrum} возьмём $h\in\Phi_d$ с $\Sp(h,P)=\Sp(P)$ и окрестность $V$ из леммы~\ref{lem:stability}. При $Q\in V$ по лемме~\ref{lem:necessity} $\Sp(P)=\Sp(h,P)\subseteq\Sp(h,Q)\subseteq\Sp(Q)$.
\end{proof}

\begin{remark}
    В частности, из теоремы~\ref{thm:solvability} следует, что если $y$ разрешим при ценах $P$, то $y$ разрешим при всех наборах цен из $V$.
\end{remark}

\begin{definition}[Слабая разрешимость]\label{def:weak-solvability}
    Пусть $P\in\Rp[d\times T]$~--- набор цен. Будем говорить, что вектор выпусков $y\in\Rp[T]$ \textit{слабо разрешим} при ценах $P$, если в любой открытой окрестности $U\ni P$ найдётся набор цен $Q\in U$, при котором $y$ разрешим.
\end{definition}

\begin{definition}[Слабо достижимые спектры]\label{def:weak}
    Пусть $P:=\set{p_{t}\in\Rp[d]}_{t\in[T]}$~--- набор цен. Будем говорить, что $S\subset[T]$~--- \textit{слабо достижимый спектр} при ценах $P$, если для каждого $t\in S$ существует ненулевой вектор $\xi\in\Rp[d]$, для которого $\ip{\xi}{p_s-p_t}\ge0$ при всех $s\notin S$. Такой вектор назовём \textit{слабым свидетелем} индекса $t$. Будем обозначать как $\widehat{\Sp}(P)$ семейство таких спектров.
    Отметим, что $\Sp(P)\subseteq\widehat{\Sp}(P)$.
\end{definition}

\begin{lemma}\label{lem:limit}
    Пусть $P\in\Rp[d\times T]$. Тогда существует такая окрестность $U\ni P$, $U\subset\Rp[d\times T]$, что для каждого набора цен $Q\in U$ выполнено $\Sp(Q)\subseteq\widehat{\Sp}(P)$.
\end{lemma}

\begin{proof}
    Иначе, поскольку спектров конечное число, нашлись бы $S\notin\widehat{\Sp}(P)$, $t\in S$ и наборы $Q_n\to P$ с $S\in\Sp(Q_n)$. Свидетелей $\xi_n$ индекса $t$ при ценах $Q_n$ нормируем условием $\ip{\xi_n}{\mathbf e}=1$. Симплекс компактен, и предел $\xi$ подпоследовательности ненулевой, неотрицательный и удовлетворяет $\ip{\xi}{p_s-p_t}\ge0$ при $s\notin S$, то есть является слабым свидетелем. Это противоречит выбору $S$ и $t$.
\end{proof}

\begin{lemma}[перепаривание]\label{lem:rematch}
    Пусть $C\subset[T]$~--- набор индексов с одинаковыми ценами, $u\in\widehat{\Sp}(P)$, $u\cap C\ne\varnothing$, $C\setminus u\ne\varnothing$ и $L\subset C$. Тогда $(u\setminus C)\cup L\in\widehat{\Sp}(P)$.
\end{lemma}

\begin{proof}
    Пусть $S':=(u\setminus C)\cup L$. Для $t\in u\setminus C$ подходит его слабый свидетель для $u$: индекс $s\notin S'$ вне $C$ лежит вне $u$, а индекс $s\in C\setminus L$ имеет ту же цену, что некоторый $s_1\in C\setminus u$. Для $t\in L$ подходит слабый свидетель любого $t_1\in u\cap C$ для $u$: цены $p_t$ и $p_{t_1}$ равны, индексы $s\notin S'$ вне $C$ лежат вне $u$, а для $s\in C$ разность $p_s-p_t$ равна нулю.
\end{proof}

Для $S\subset[T]$ обозначим через $c(S)$ число пар $(a,b)\in S\times S$ с $p_a=p_b$.

\begin{lemma}[распутывание]\label{lem:uncross}
    Пусть $u,u'\in\widehat{\Sp}(P)$, $t\in u\setminus u'$, $s\in u'\setminus u$ и $p_s=p_t$. Тогда найдутся $A,B\in\widehat{\Sp}(P)$, для которых $\mathbf 1_A+\mathbf 1_B=\mathbf 1_u+\mathbf 1_{u'}$ и $c(A)+c(B)>c(u)+c(u')$.
\end{lemma}

\begin{proof}
    Пусть $C:=\set{r\in[T]\mid p_r=p_t}$. Положим
    \[
        A:=(u\setminus C)\cup\bigl((u\cup u')\cap C\bigr),\qquad B:=(u'\setminus C)\cup(u\cap u'\cap C).
    \]
    Множество $u$ пересекает $C$ по $t$ и не содержит $s\in C$, а $u'$ пересекает $C$ по $s$ и не содержит $t$. Поэтому $A,B\in\widehat{\Sp}(P)$ по лемме~\ref{lem:rematch}. Вне $C$ множество $A$ совпадает с $u$, а $B$ с $u'$. На $C$ они дают объединение и пересечение $u\cap C$ и $u'\cap C$, откуда $\mathbf 1_A+\mathbf 1_B=\mathbf 1_u+\mathbf 1_{u'}$.
    Пара $(a,b)$ с $p_a=p_b$ лежит целиком в $C$ или целиком вне $C$. Вне $C$ ничего не меняется. Пара из $C$, лежащая в $u$ или в $u'$, лежит в $A$, а лежащая в обоих~--- ещё и в $B$. Поэтому ни одна пара не теряется. Пара $(t,s)$ лежит в $A$, но не лежит ни в $u$, ни в $u'$.
\end{proof}

\begin{lemma}[монотонное представление]\label{lem:monotone}
    Пусть $y\in\cone\set{\mathbf 1_S\mid S\in\widehat{\Sp}(P)}$. Тогда $y=\sum_Sm_S\mathbf 1_S$ при некоторых $m_S\ge0$, $S\in\widehat{\Sp}(P)$, причём для всякого $S$ с $m_S>0$ из $t\in S$, $s\notin S$ и $p_s=p_t$ следует $y_s<y_t$.
\end{lemma}

\begin{proof}
    Пусть $y=\sum_Sm^0_S\mathbf 1_S$~--- какое-нибудь представление и $N:=\sum_Sm^0_S$. Представления $y=\sum_Sm_S\mathbf 1_S$ с $m_S\ge0$, $m_S=0$ при $S\notin\widehat{\Sp}(P)$ и $\sum_Sm_S\le N$ образуют непустой компакт. Возьмём на нём представление, на котором $\sum_Sm_Sc(S)$ максимальна.
    Пусть $m_u,m_{u'}>0$ и нашлись $t\in u\setminus u'$, $s\in u'\setminus u$ с $p_s=p_t$. Перенесём вес $\varepsilon:=\min(m_u,m_{u'})$ с $u$ и $u'$ на множества $A$ и $B$ из леммы~\ref{lem:uncross}. Представление и полный вес сохраняются, а $\sum_Sm_Sc(S)$ вырастает на $\varepsilon\bigl(c(A)+c(B)-c(u)-c(u')\bigr)>0$. Это противоречит выбору.
    Значит, если $m_S>0$, $t\in S$, $s\notin S$ и $p_s=p_t$, то всякое $U$ с $m_U>0$, содержащее $s$, содержит и $t$. Поэтому все слагаемые в $y_t-y_s=\sum_Um_U\bigl(\mathbf 1_U(t)-\mathbf 1_U(s)\bigr)$ неотрицательны, а слагаемое с $U=S$ равно $m_S>0$.
\end{proof}

\begin{remark}
    Исправлять представление необходимо. При $p_1=p_2=(1,1)$, $p_3=(1,\tfrac12)$, $p_4=(\tfrac12,1)$, $p_5=(\tfrac12,\tfrac32)$, $p_6=(\tfrac32,\tfrac12)$ спектры $\set{1,4,5}$, $\set{1,3,6}$, $\set{2,3,4}$ слабо достижимы, но ни при каких ценах, близких к $P$, не достижимы одновременно.
\end{remark}

\begin{lemma}[наклон]\label{lem:tilt}
    Пусть $P:=\set{p_t\in\Rp[d]}_{t\in[T]}$~--- набор цен, $y\in\Rp[T]$, и при $\delta>0$ положим
    \[
        Q_\delta:=\set{q_t}_{t\in[T]},\qquad q_t:=p_t+\delta\Bigl(\sum_{i\in[d]}e^{-p^i_t}-\delta y_t\Bigr)\mathbf e.
    \]
    Тогда $Q_\delta\in\Rpp[d\times T]$ при малых $\delta>0$. Если $S\in\widehat{\Sp}(P)$ и $y_s<y_t$ при всех $t\in S$, $s\notin S$ с $p_s=p_t$, то $S\in\Sp(Q_\delta)$ при всех достаточно малых $\delta>0$.
\end{lemma}

\begin{proof}
    Пусть $F_r:=\sum_ie^{-p^i_r}>0$. Тогда $q^i_t\ge\delta(F_t-\delta y_t)>0$ при малых $\delta$.
    Пусть $t\in S$, $\xi$~--- слабый свидетель индекса $t$ с $\ip{\xi}{\mathbf e}=1$, $g:=(e^{-p^1_t},\dots,e^{-p^d_t})$ и $\xi_\delta:=\xi+\delta g$. Тогда $q_s-q_t=(p_s-p_t)+\bigl(\delta(F_s-F_t)-\delta^2(y_s-y_t)\bigr)\mathbf e$. Покажем, что $\ip{\xi_\delta}{q_s-q_t}>0$ при $s\notin S$ и малых $\delta$.
    Если $p_s=p_t$, то $\ip{\xi_\delta}{q_s-q_t}=\delta^2(y_t-y_s)\bigl(1+\delta\ip{g}{\mathbf e}\bigr)>0$.
    Если $p_s\ne p_t$, то $\ip{\xi_\delta}{q_s-q_t}=\ip{\xi}{p_s-p_t}+\delta K+O(\delta^2)$ с $K:=F_s-F_t+\ip{g}{p_s-p_t}=\sum_ie^{-p^i_t}\bigl(e^{-x_i}-1+x_i\bigr)$, $x_i:=p^i_s-p^i_t$. Здесь $\ip{\xi}{p_s-p_t}\ge0$, а $K>0$, так как $e^{-x}\ge1-x$ с равенством только при $x=0$.
    Пар $(t,s)$ конечное число, и $\xi_\delta\ge\delta g$ ненулевой, так что $\xi_\delta$~--- свидетель индекса $t$ при ценах $Q_\delta$.
\end{proof}

\begin{theorem}[критерий слабой разрешимости]\label{thm:closure}
    Пусть $P\in\Rp[d\times T]$~--- набор цен. Вектор выпусков $y\in\Rp[T]$ слабо разрешим при ценах $P$ тогда и только тогда, когда $y\in\cone\set{\mathbf 1_S\mid S\in\widehat{\Sp}(P)}$.
\end{theorem}

\begin{proof}
    Пусть $y$ слабо разрешим. По лемме~\ref{lem:limit} в некоторой окрестности $P$ выполнено $\Sp(Q)\subseteq\widehat{\Sp}(P)$, и в ней есть $Q$, при котором $y$ разрешим. По теореме~\ref{thm:solvability} $y\in\cone\set{\mathbf 1_S\mid S\in\Sp(Q)}\subseteq\cone\set{\mathbf 1_S\mid S\in\widehat{\Sp}(P)}$.
    Обратно, возьмём представление из леммы~\ref{lem:monotone}. По лемме~\ref{lem:tilt} все его спектры, которых конечное число, лежат в $\Sp(Q_\delta)$ при малых $\delta>0$, и $Q_\delta\in\Rpp[d\times T]$. Тогда $y$ разрешим при ценах $Q_\delta$ по теореме~\ref{thm:solvability}, а $Q_\delta\to P$ при $\delta\to0$, так что $Q_\delta$ попадает в любую окрестность $P$.
\end{proof}
